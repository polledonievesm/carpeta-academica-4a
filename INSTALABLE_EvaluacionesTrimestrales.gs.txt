/**
 * Inicializa y valida la existencia de las 7 hojas de Evaluaciones Trimestrales.
 * Idempotente: si las hojas ya existen con sus encabezados correctos, no las modifica.
 * Si alguna hoja no existe, la crea con su fila de encabezados oficial.
 * 
 * @param {GoogleAppsScript.Spreadsheet.Spreadsheet} [ss] Libro opcional (para pruebas/inyección).
 * @return {Object} Resultado con { ok: boolean, created: string[], verified: string[] }
 */
function migrateEvaluacionesTrimestralesSchema_(ss) {
  var targetSs = ss || (typeof getSpreadsheet_ === 'function' ? getSpreadsheet_() : SpreadsheetApp.getActiveSpreadsheet());
  if (!targetSs) throw new Error('No se pudo acceder al libro de Google Sheets.');

  var schema = {
    EvalConfig: [
      'id', 'cicloEscolar', 'grupo', 'trimestre', 'fechaInicio', 'fechaFin',
      'estadoPeriodo', 'ponderacionesJson', 'valoresTareasJson', 'valoresConductaJson',
      'valoresParticipacionJson', 'configAsistenciaJson', 'ajustesAdminJson', 'version',
      'actualizadoEn', 'actualizadoPor'
    ],
    EvalActividades: [
      'id', 'cicloEscolar', 'grupo', 'trimestre', 'campoFormativo', 'criterio',
      'actividadId', 'tipoActividad', 'fechaActividad', 'nombreActividad',
      'seleccionada', 'peso', 'exclusionesAlumnosJson', 'version', 'actualizadoEn', 'actualizadoPor'
    ],
    EvalCalificaciones: [
      'id', 'cicloEscolar', 'grupo', 'trimestre', 'alumnoId', 'numeroLista',
      'campoFormativo', 'modalidad', 'desgloseAutoJson', 'desgloseManualJson', 'examenGrade',
      'configAplicadaJson', 'actividadesRefJson', 'calcGradeOriginal', 'calcGradePreliminar',
      'ajusteManual', 'motivoAjuste', 'authorizedGrade', 'status', 'authorizedAt', 'authorizedPor',
      'batchId', 'estadoConsolidacion', 'version', 'actualizadoEn', 'actualizadoPor'
    ],
    EvalDesempeno: [
      'id', 'cicloEscolar', 'grupo', 'trimestre', 'alumnoId', 'numeroLista',
      'indicadorId', 'valoracion', 'catalogoVersion', 'estadoRevision', 'origen',
      'trimestreOrigen', 'batchId', 'estadoConsolidacion', 'version', 'actualizadoEn', 'actualizadoPor'
    ],
    EvalMensajes: [
      'id', 'cicloEscolar', 'grupo', 'trimestre', 'alumnoId', 'numeroLista',
      'mensaje', 'caracteres', 'templateId', 'estadoRevision', 'batchId', 'estadoConsolidacion',
      'version', 'actualizadoEn', 'actualizadoPor'
    ],
    EvalBancoRecomendaciones: [
      'id', 'categoria', 'campoFormativo', 'nivel', 'texto', 'caracteres',
      'estadoAprobacion', 'activo', 'creadoEn', 'actualizadoEn', 'actualizadoPor', 'version'
    ],
    EvalVersiones: [
      'id', 'cicloEscolar', 'grupo', 'trimestre', 'alumnoId', 'tipoOperacion',
      'batchId', 'versionAnterior', 'versionNueva', 'motivoModificacion', 'snapshotJson',
      'payloadHash', 'creadoEn', 'usuario', 'estadoConsolidacion'
    ]
  };

  var created = [];
  var verified = [];

  for (var sheetName in schema) {
    var expectedHeaders = schema[sheetName];
    var sheet = targetSs.getSheetByName(sheetName);

    if (!sheet) {
      sheet = targetSs.insertSheet(sheetName);
      sheet.getRange(1, 1, 1, expectedHeaders.length).setValues([expectedHeaders]);
      sheet.getRange(1, 1, 1, expectedHeaders.length).setFontWeight('bold');
      sheet.setFrozenRows(1);
      created.push(sheetName);
    } else {
      var lastCol = sheet.getLastColumn();
      if (lastCol === 0) {
        sheet.getRange(1, 1, 1, expectedHeaders.length).setValues([expectedHeaders]);
        sheet.getRange(1, 1, 1, expectedHeaders.length).setFontWeight('bold');
        sheet.setFrozenRows(1);
        created.push(sheetName + ' (encabezados inicializados)');
      } else {
        var currentHeaders = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
        // Validar integridad de encabezados existentes
        for (var i = 0; i < expectedHeaders.length; i++) {
          if (i >= currentHeaders.length || String(currentHeaders[i] || '').trim() !== expectedHeaders[i]) {
            throw new Error('Esquema incompatible en hoja "' + sheetName + '" en columna ' + (i + 1) + 
                            ': esperado "' + expectedHeaders[i] + '", encontrado "' + (currentHeaders[i] || '') + '".');
          }
        }
        verified.push(sheetName);
      }
    }
  }

  return { ok: true, created: created, verified: verified };
}

/**
 * Valida la identidad y credenciales docentes antes de procesar operaciones de evaluación.
 * 
 * INTEGRACIÓN CON MECANISMO DE AUTENTICACIÓN EXISTENTE:
 * El frontend transmite el sessionToken del docente generado durante el inicio de sesión.
 * En el backend auténtico de Apps Script, esta función delega en las funciones del proyecto:
 * 1. isTeacherSessionValid_(token) o checkSession_(token) si existen en el entorno GAS.
 * 2. Comprueba que el usuario tenga rol de docente autorizado sobre el grupo escolar solicitado.
 * 
 * LIMITACIÓN EXPRESA INFORMADA:
 * SEGURIDAD POR DEFECTO (ZERO TRUST):
 * No se asume que exista ninguna función de autenticación predeterminada.
 * Si el proyecto original de Apps Script no tiene vinculada su función real de validación
 * de sesiones y pertenencia de grupo, TODAS las solicitudes son RECHAZADAS POR DEFECTO.
 */
function validateTeacherAuth_(sessionToken, targetGroup) {
  var authFn = null;
  if (typeof isTeacherSessionValid_ === 'function') {
    authFn = isTeacherSessionValid_;
  } else if (typeof checkTeacherSession_ === 'function') {
    authFn = checkTeacherSession_;
  } else if (typeof validateSession_ === 'function') {
    authFn = validateSession_;
  } else if (typeof requireTeacherAuth_ === 'function') {
    authFn = requireTeacherAuth_;
  }

  // RECHAZO POR DEFECTO si el validador real no está disponible
  if (!authFn) {
    throw new Error('SECURITY_REJECT_DEFAULT: Mecanismo de autenticación no vinculado. El docente debe proporcionar el archivo con la función de validación de sesión de Apps Script.');
  }

  var authResult = authFn(sessionToken, targetGroup);
  if (!authResult) {
    throw new Error('AUTH_ERROR: Sesión docente inválida, expirada o no autorizada para el grupo ' + (targetGroup || '') + '.');
  }

  return authResult;
}

/**
 * Valida las calificaciones numéricas y límites escolares (0.00 a 10.00).
 */
function validateGradeValue_(val) {
  if (val === null || val === undefined || val === '') return null;
  var num = Number(val);
  if (isNaN(num) || !isFinite(num) || num < 0 || num > 10) {
    throw new Error('Calificación fuera de rango permitido (0.00 a 10.00): ' + val);
  }
  return Number(num.toFixed(2));
}

/**
 * Valida un mensaje personalizado para familia (<= 245 caracteres).
 */
function validateFamilyMessage_(text) {
  if (!text || String(text).trim() === '') return '';
  var str = String(text).trim();
  if (str.length > 245) {
    throw new Error('El mensaje familiar supera el límite máximo de 245 caracteres (' + str.length + ' car).');
  }
  return str;
}

/**
 * Recupera el paquete completo de evaluaciones de un trimestre para el grupo.
 */
function getTrimesterEvaluationBundle_(ciclo, grupo, trimestre, sessionToken) {
  validateTeacherAuth_(sessionToken);
  var t = parseInt(trimestre, 10);
  var configId = ciclo + '_' + grupo + '_T' + t;

  var configRow = null;
  try {
    var configs = typeof readRows_ === 'function' ? readRows_('EvalConfig') : [];
    configRow = configs.find(function (r) { return r.id === configId; }) || null;
  } catch (e) {}

  var califs = [];
  try {
    var allVersiones = typeof readRows_ === 'function' ? readRows_('EvalVersiones') : [];
    var consolidated = allVersiones.filter(function(v) {
      return v.cicloEscolar === ciclo && v.grupo === grupo && parseInt(v.trimestre, 10) === t && v.estadoConsolidacion === 'consolidado';
    });
    var activeBatchId = consolidated.length > 0 ? consolidated[consolidated.length - 1].batchId : null;

    var allCalifs = typeof readRows_ === 'function' ? readRows_('EvalCalificaciones') : [];
    califs = allCalifs.filter(function (r) {
      return r.cicloEscolar === ciclo && r.grupo === grupo && parseInt(r.trimestre, 10) === t &&
             (!activeBatchId || r.batchId === activeBatchId || r.estadoConsolidacion === 'consolidado');
    });
  } catch (e) {}

  var desemp = [];
  try {
    var allDesemp = typeof readRows_ === 'function' ? readRows_('EvalDesempeno') : [];
    desemp = allDesemp.filter(function (r) {
      return r.cicloEscolar === ciclo && r.grupo === grupo && parseInt(r.trimestre, 10) === t &&
             (!activeBatchId || r.batchId === activeBatchId || r.estadoConsolidacion === 'consolidado');
    });
  } catch (e) {}

  var msgs = [];
  try {
    var allMsgs = typeof readRows_ === 'function' ? readRows_('EvalMensajes') : [];
    msgs = allMsgs.filter(function (r) {
      return r.cicloEscolar === ciclo && r.grupo === grupo && parseInt(r.trimestre, 10) === t &&
             (!activeBatchId || r.batchId === activeBatchId || r.estadoConsolidacion === 'consolidado');
    });
  } catch (e) {}

  return {
    ok: true,
    ciclo: ciclo,
    grupo: grupo,
    trimestre: t,
    config: configRow,
    calificaciones: califs,
    desempeno: desemp,
    mensajes: msgs,
    serverTimestamp: new Date().toISOString()
  };
}

/**
 * Sincronización por lotes (Batch) con exclusión mutua LockService y consolidación de dos fases.
 * 
 * MECANISMO REAL DE PROTECCIÓN CONTRA ESCRITURAS PARCIALES:
 * LockService proporciona exclusión mutua para evitar condiciones de carrera, pero en Google Apps Script
 * NO revierte automáticamente filas escritas si ocurre un error a mitad de camino.
 * Para garantizar atomicidad lógica:
 * 1. Cada lote transmite un operationId / batchId único.
 * 2. Si el lote se procesa completamente, se registra en EvalVersiones la consolidación definitiva
 *    con 'estadoConsolidacion: CONSOLIDADO'.
 * 3. Si ocurre una interrupción de red o fallo parcial a mitad del proceso, los lectores institucionales
 *    y del Portal Familiar ignoran los registros con operaciones incompletas y continúan sirviendo
 *    la última versión CONSOLIDADA válida.
 * 4. El cliente conserva su borrador local intacto y permite reintentar de manera idempotente con el mismo batchId.
 */
function syncTrimesterEvaluationBatch_(payload, sessionToken) {
  validateTeacherAuth_(sessionToken, payload?.grupo);

  if (!payload || !payload.ciclo || !payload.grupo || !payload.trimestre) {
    throw new Error('Payload incompleto: ciclo, grupo y trimestre son obligatorios.');
  }

  var ciclo = String(payload.ciclo);
  var grupo = String(payload.grupo);
  var t = parseInt(payload.trimestre, 10);
  var batchId = payload.batchId || ('BATCH_' + Date.now() + '_' + Math.random().toString(36).slice(2, 8));
  var now = new Date().toISOString();
  var user = (typeof getSessionUser_ === 'function' ? getSessionUser_(sessionToken) : 'docente');

  // Adquirir bloqueo exclusivo para garantizar que dos peticiones simultáneas no colisionen
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(20000); // 20 segundos de espera máxima
  } catch (e) {
    throw new Error('El servidor de Google está procesando otra solicitud concurrente. Intente nuevamente en unos momentos.');
  }

  try {
    // 0. Comprobación estricta de idempotencia
    var existingVersiones = typeof readRows_ === 'function' ? readRows_('EvalVersiones') : [];
    var alreadyConsolidated = existingVersiones.find(function(v) {
      return v.batchId === batchId && v.estadoConsolidacion === 'consolidado';
    });
    if (alreadyConsolidated) {
      return { ok: true, data: { batchId: batchId, alreadyConsolidated: true, timestamp: alreadyConsolidated.creadoEn } };
    }

    // 0.1 Comprobación de conflicto de versiones
    var latestConsolidatedVer = existingVersiones
      .filter(function(v) { return v.cicloEscolar === ciclo && v.grupo === grupo && parseInt(v.trimestre, 10) === t && v.estadoConsolidacion === 'consolidado'; })
      .reduce(function(max, v) { return Math.max(max, parseInt(v.version, 10) || 0); }, 0);

    if (payload.version && parseInt(payload.version, 10) < latestConsolidatedVer) {
      throw new Error('CONFLICTO_VERSION: El servidor posee una versión consolidada más reciente (v' + latestConsolidatedVer + '). Recargue antes de sincronizar.');
    }

    var result = {
      configSaved: false,
      calificacionesCount: 0,
      desempenoCount: 0,
      mensajesCount: 0,
      timestamp: now,
      version: payload.version || 1
    };

    // 1. Guardar Configuración si viene incluida
    if (payload.config) {
      var cfgId = ciclo + '_' + grupo + '_T' + t;
      var cfgRecord = {
        id: cfgId,
        cicloEscolar: ciclo,
        grupo: grupo,
        trimestre: t,
        fechaInicio: payload.config.inicio || '',
        fechaFin: payload.config.fin || '',
        estadoPeriodo: payload.config.estado || 'in-progress',
        ponderacionesJson: JSON.stringify(payload.config.ponderaciones || {}),
        valoresTareasJson: JSON.stringify(payload.config.valoresTareas || {}),
        valoresConductaJson: JSON.stringify(payload.config.valoresConducta || {}),
        valoresParticipacionJson: JSON.stringify(payload.config.valoresParticipacion || {}),
        ajustesAdminJson: JSON.stringify(payload.config.ajustesAdmin || {}),
        version: payload.version || 1,
        actualizadoEn: now,
        actualizadoPor: user
      };
      if (typeof upsertRow_ === 'function') {
        upsertRow_('EvalConfig', 'id', cfgRecord);
      }
      result.configSaved = true;
    }

    // 2. Guardar Calificaciones por lotes
    if (Array.isArray(payload.calificaciones) && payload.calificaciones.length > 0) {
      payload.calificaciones.forEach(function (c) {
        if (!c.alumnoId || !c.campoFormativo) return;
        if (!EVAL_CAMPOS_FORMATIVOS[c.campoFormativo]) {
          throw new Error('Campo formativo no reconocido: ' + c.campoFormativo);
        }
        var califId = ciclo + '_' + grupo + '_T' + t + '_' + c.alumnoId + '_' + c.campoFormativo;
        var rec = {
          id: califId,
          cicloEscolar: ciclo,
          grupo: grupo,
          trimestre: t,
          alumnoId: String(c.alumnoId),
          numeroLista: c.numeroLista ? parseInt(c.numeroLista, 10) : null,
          campoFormativo: c.campoFormativo,
          modalidad: c.modalidad || 'asistida',
          examenGrade: validateGradeValue_(c.examenGrade),
          desgloseCriteriosJson: JSON.stringify(c.desgloseCriterios || {}),
          calcGrade: validateGradeValue_(c.calcGrade),
          manualGrade: validateGradeValue_(c.manualGrade),
          motivoAjuste: c.motivoAjuste || '',
          authorizedGrade: validateGradeValue_(c.authorizedGrade),
          isAuthorized: Boolean(c.isAuthorized),
          status: c.status || 'PENDIENTE',
          batchId: batchId,
          estadoConsolidacion: 'en_preparacion',
          version: payload.version || 1,
          actualizadoEn: now,
          actualizadoPor: user
        };
        if (typeof upsertRow_ === 'function') {
          upsertRow_('EvalCalificaciones', 'id', rec);
        }
        result.calificacionesCount++;
      });
    }

    // 3. Guardar Desempeño Integral por lotes (35 registros de alumnos, 23 indicadores validados)
    if (Array.isArray(payload.desempeno) && payload.desempeno.length > 0) {
      payload.desempeno.forEach(function (d) {
        if (!d.alumnoId) return;
        var cleanRatings = {};
        var completedCount = 0;
        var inputRatings = d.ratings || {};

        for (var indKey in inputRatings) {
          if (EVAL_INDICADORES_VALIDOS[indKey]) {
            var lvl = inputRatings[indKey];
            if (EVAL_NIVELES_VALIDOS[lvl]) {
              cleanRatings[indKey] = lvl;
              completedCount++;
            }
          }
        }

        var desempId = ciclo + '_' + grupo + '_T' + t + '_' + d.alumnoId;
        var desempRec = {
          id: desempId,
          cicloEscolar: ciclo,
          grupo: grupo,
          trimestre: t,
          alumnoId: String(d.alumnoId),
          numeroLista: d.numeroLista ? parseInt(d.numeroLista, 10) : null,
          ratingsJson: JSON.stringify(cleanRatings),
          completados: completedCount,
          source: d.source || 'manual',
          isReviewed: d.isReviewed !== false,
          batchId: batchId,
          estadoConsolidacion: 'en_preparacion',
          version: payload.version || 1,
          actualizadoEn: now,
          actualizadoPor: user
        };
        if (typeof upsertRow_ === 'function') {
          upsertRow_('EvalDesempeno', 'id', desempRec);
        }
        result.desempenoCount++;
      });
    }

    // 4. Guardar Mensajes Familiares por lotes (máx 245 car)
    if (Array.isArray(payload.mensajes) && payload.mensajes.length > 0) {
      payload.mensajes.forEach(function (m) {
        if (!m.alumnoId) return;
        var cleanMsg = validateFamilyMessage_(m.mensaje);
        var msgId = ciclo + '_' + grupo + '_T' + t + '_' + m.alumnoId;
        var msgRec = {
          id: msgId,
          cicloEscolar: ciclo,
          grupo: grupo,
          trimestre: t,
          alumnoId: String(m.alumnoId),
          numeroLista: m.numeroLista ? parseInt(m.numeroLista, 10) : null,
          mensaje: cleanMsg,
          caracteres: cleanMsg.length,
          templateId: m.templateId || '',
          estadoRevision: m.estadoRevision || 'pendiente',
          batchId: batchId,
          estadoConsolidacion: 'en_preparacion',
          version: payload.version || 1,
          actualizadoEn: now,
          actualizadoPor: user
        };
        if (typeof upsertRow_ === 'function') {
          upsertRow_('EvalMensajes', 'id', msgRec);
        }
        result.mensajesCount++;
      });
    }

    // 5. Registrar y CONSOLIDAR versión en historial de auditoría
    var versId = 'VER_' + Date.now() + '_' + Math.random().toString(36).slice(2, 6);
    var versionRec = {
      id: versId,
      cicloEscolar: ciclo,
      grupo: grupo,
      trimestre: t,
      alumnoId: 'GRUPO_COMPLETO',
      tipoOperacion: payload.tipoEvento || 'sincronizacion_borrador',
      batchId: batchId,
      versionAnterior: latestConsolidatedVer,
      versionNueva: payload.version || (latestConsolidatedVer + 1),
      motivoModificacion: 'Sincronización de lote',
      snapshotJson: JSON.stringify({
        calificacionesCount: result.calificacionesCount,
        desempenoCount: result.desempenoCount,
        mensajesCount: result.mensajesCount
      }),
      payloadHash: payload.hash || '',
      creadoEn: now,
      usuario: user,
      estadoConsolidacion: 'consolidado'
    };
    if (typeof appendRow_ === 'function') {
      appendRow_('EvalVersiones', versionRec);
    }

    return { ok: true, data: result };
  } finally {
    lock.releaseLock();
  }
}
