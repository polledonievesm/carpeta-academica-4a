/**
 * TÍTULO: Validación de seguridad docente con política Zero-Trust
 * NOMBRE TÉCNICO: validateTeacherAuth_
 * FECHA DE CREACIÓN: 27/11/2026
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * 
 * QUÉ HACE:
 *   Verifica que la sesión docente exista, esté vigente y tenga permisos sobre el grupo escolar.
 *   NO realiza conjeturas de funciones ni selecciona nombres especulativos.
 *   Si la función auténtica del proyecto no está vinculada, RECHAZA POR DEFECTO.
 * 
 * PARA QUÉ SIRVE:
 *   Garantiza que ninguna petición no autorizada o manipulada pueda consultar ni alterar
 *   calificaciones de los alumnos. Provee además la identidad responsable autenticada.
 * 
 * POR QUÉ SE CREÓ:
 *   Para cumplir la política de seguridad Zero-Trust requerida por la dirección escolar.
 * 
 * PARÁMETROS:
 *   @param {string} sessionToken Token de sesión transmitido en la petición.
 *   @param {string} targetGroup Grupo escolar solicitado (ej. '4A').
 * 
 * RESULTADO:
 *   @return {Object} Objeto de sesión validado con estructura { ok: true, usuario: string, rol: string }.
 *   @throws {Error} Si el validador no está disponible o la sesión es inválida.

 * DEPENDENCIAS:
 *   validarSesion_ (Función auténtica provista por el docente).

 * HISTORIAL DE MODIFICACIONES:

/**
 * TÍTULO: Adaptador de autenticación docente
 * FECHA DE MODIFICACIÓN: 27/11/2026
 * PROPÓSITO Y MOTIVO: Reutiliza validarSesion_ y limita el ámbito al grupo y ciclo configurados; devuelve identidad del servidor.
 * HISTORIAL: 27/11/2026 — integración focalizada 7A.3.
 */
/**
 * TÍTULO: Normalización canónica del identificador de grupo escolar
 * NOMBRE TÉCNICO: evalNormalizeGroup_
 * FECHA DE CREACIÓN: 27/11/2026
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * 
 * QUÉ HACE:
 *   Elimina espacios, caracteres de grado (°, º) y convierte a mayúsculas.
 * 
 * PARA QUÉ SIRVE:
 *   Garantiza que '4°A', '4° A', '4A' y '4 A' se resuelvan de forma idéntica ('4A').
 * 
 * PARÁMETROS:
 *   @param {string} g Cadena con el identificador del grupo.
 * 
 * RESULTADO:
 *   @return {string} Identificador normalizado en mayúsculas (ej. '4A').
 */
function evalNormalizeGroup_(g) {
  return String(g || '').replace(/[°º\s]/g, '').toUpperCase();
}

/**
 * TÍTULO: Validación estricta del número de trimestre escolar
 * NOMBRE TÉCNICO: evalValidateTrimester_
 * FECHA DE CREACIÓN: 27/11/2026
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * 
 * QUÉ HACE:
 *   Verifica que el trimestre sea un número entero y pertenezca exclusivamente al conjunto {1, 2, 3}.
 * 
 * PARA QUÉ SIRVE:
 *   Impide la inserción o consulta de períodos inexistentes o malformados.
 * 
 * PARÁMETROS:
 *   @param {*} trimestre Valor del trimestre recibido.
 * 
 * RESULTADO:
 *   @return {number} Número entero validado (1, 2 o 3).
 *   @throws {Error} Si el trimestre no es 1, 2 o 3.
 */
function evalValidateTrimester_(trimestre) {
  var t = parseInt(trimestre, 10);
  if (![1, 2, 3].includes(t)) {
    throw new Error('TRIMESTRE_INVALIDO: El trimestre debe ser exclusivamente 1, 2 o 3 (recibido: ' + trimestre + ').');
  }
  return t;
}

/**
 * TÍTULO: Obtención del padrón escolar autorizado de alumnos
 * NOMBRE TÉCNICO: evalGetAuthorizedStudentsMap_
 * FECHA DE CREACIÓN: 27/11/2026
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * 
 * QUÉ HACE:
 *   Consulta la tabla oficial 'Alumnos' y construye un diccionario de alumnos activos indexado por id.
 * 
 * PARA QUÉ SIRVE:
 *   Valida que los registros de evaluación pertenezcan a alumnos oficialmente inscritos en el aula.
 * 
 * PARÁMETROS:
 *   @param {string} normalizedGroup Identificador normalizado del grupo.
 * 
 * RESULTADO:
 *   @return {Object} Diccionario { [alumnoId]: studentObject }.
 */
function evalGetAuthorizedStudentsMap_(normalizedGroup) {
  var map = {};
  var students = (typeof readRows_ === 'function') ? readRows_('Alumnos') : [];
  if (Array.isArray(students)) {
    students.forEach(function (s) {
      if (s && s.id && s.activo !== false && String(s.activo).toLowerCase() !== 'false') {
        map[String(s.id)] = s;
      }
    });
  }
  return map;
}

/**
 * TÍTULO: Validación de pertenencia de alumno al padrón autorizado
 * NOMBRE TÉCNICO: evalValidateStudentId_
 * FECHA DE CREACIÓN: 27/11/2026
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * 
 * QUÉ HACE:
 *   Verifica que el alumnoId exista y esté activo en el padrón escolar del aula.
 * 
 * PARA QUÉ SIRVE:
 *   Bloquea la inserción de identificadores de alumnos inexistentes o no autorizados.
 * 
 * PARÁMETROS:
 *   @param {*} alumnoId Identificador del alumno a validar.
 *   @param {Object} authorizedMap Diccionario del padrón autorizado.
 *   @param {string} normalizedGroup Grupo escolar normalizado.
 * 
 * RESULTADO:
 *   @throws {Error} Si el alumno no pertenece al padrón oficial activo.
 */
function evalValidateStudentId_(alumnoId, authorizedMap, normalizedGroup) {
  var idStr = String(alumnoId || '').trim();
  if (!idStr) {
    throw new Error('ALUMNO_INVALIDO: Falta el identificador de alumno.');
  }
  if (authorizedMap && Object.keys(authorizedMap).length > 0) {
    if (!authorizedMap[idStr]) {
      throw new Error('ALUMNO_NO_AUTORIZADO: El alumno "' + idStr + '" no pertenece al padrón escolar autorizado del grupo ' + normalizedGroup + '.');
    }
  }
}

function validateTeacherAuth_(sessionToken, targetGroup, ciclo) {
  var session = validarSesion_(sessionToken);
  var gExpected = evalNormalizeGroup_(targetGroup);
  var gSession = evalNormalizeGroup_(session ? session.grupo : '');
  if (!session || !session.ok || !session.usuario || gSession !== gExpected || String(session.cicloEscolar) !== String(ciclo)) {
    throw new Error('AUTH_ERROR: ámbito no autorizado para el grupo escolar ' + targetGroup + ' y ciclo ' + ciclo + '.');
  }
  return session;
}

/**
 * TÍTULO: Bloqueo de concurrencia mediante ScriptLock para peticiones Web App
 * NOMBRE TÉCNICO: evalWithScriptLock_
 * FECHA DE CREACIÓN: 27/11/2026
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * 
 * QUÉ HACE:
 *   Adquiere un bloqueo exclusivo ScriptLock con tiempo de espera configurable.
 *   Ejecuta el callback y garantiza la liberación del bloqueo en el bloque finally.
 * 
 * PARA QUÉ SIRVE:
 *   Previene colisiones y condiciones de carrera en peticiones concurrentes desde GitHub Pages
 *   sin depender de getDocumentLock (que puede ser nulo o inconsistente en Web Apps).
 * 
 * POR QUÉ SE CREÓ:
 *   Para alinear el módulo con las soluciones probadas en AttendanceMonthly y DirectoryFamily,
 *   evitando bloqueos anidados o deadlocks.
 * 
 * PARÁMETROS:
 *   @param {Function} callback Operación a ejecutar de forma protegida.
 *   @param {number} [waitMs=25000] Tiempo máximo de espera en milisegundos.
 * 
 * RESULTADO:
 *   @return {*} Resultado devuelto por el callback.
 *   @throws {Error} Si no se adquiere el bloqueo en el tiempo límite.
 * 
 * DEPENDENCIAS:
 *   LockService.getScriptLock.
 */
function evalWithScriptLock_(callback, waitMs) {
  var lock = LockService.getScriptLock();
  var timeout = waitMs || 25000;
  if (!lock.tryLock(timeout)) {
    throw new Error('El servidor escolar está procesando otra solicitud concurrente. Intente nuevamente en unos segundos.');
  }
  try {
    return callback();
  } finally {
    lock.releaseLock();
  }
}

/**
 * TÍTULO: Validación y formateo de calificaciones numéricas
 * NOMBRE TÉCNICO: validateGradeValue_
 * FECHA DE CREACIÓN: 27/11/2026
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * 
 * QUÉ HACE:
 *   Valida que un valor numérico pertenezca al rango de 0.00 a 10.00 y lo redondea a 2 decimales.
 *   Si el valor es null, undefined o cadena vacía, retorna null (no evaluado).
 * 
 * PARA QUÉ SIRVE:
 *   Distingue calificaciones legítimas de 0 de evaluaciones pendientes, evitando valores inválidos.
 * 
 * POR QUÉ SE CREÓ:
 *   Para asegurar la consistencia aritmética de los registros escolares.
 * 
 * PARÁMETROS:
 *   @param {*} val Valor a validar.
 * 
 * RESULTADO:
 *   @return {number|null} Número con dos decimales o null si no existe calificación.
 */
function validateGradeValue_(val) {
  if (val === null || val === undefined || val === '') return null;
  var num = Number(val);
  if (isNaN(num)) throw new Error('Calificación numérica inválida: ' + val);
  if (num < 0 || num > 10) throw new Error('Calificación fuera del rango escolar oficial (0.00 - 10.00): ' + val);
  return Number(num.toFixed(2));
}

/**
 * TÍTULO: Validación de longitud de mensaje a familias
 * NOMBRE TÉCNICO: validateFamilyMessage_
 * FECHA DE CREACIÓN: 27/11/2026
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * 
 * QUÉ HACE:
 *   Verifica que el mensaje dirigido al Portal Familiar no supere los 245 caracteres.
 * 
 * PARA QUÉ SIRVE:
 *   Garantiza que el texto encaje en la boleta de seguimiento escolar y mensajes SMS/Portal.
 * 
 * POR QUÉ SE CREÓ:
 *   Para cumplir la restricción de diseño de reportes a familias.
 * 
 * PARÁMETROS:
 *   @param {string} msg Mensaje redactado por el docente.
 * 
 * RESULTADO:
 *   @return {string} Texto limpio y validado.
 *   @throws {Error} Si excede 245 caracteres.
 */
function validateFamilyMessage_(msg) {
  if (msg === null || msg === undefined) return '';
  var clean = String(msg).trim();
  if (clean.length > 245) {
    throw new Error('El mensaje familiar excede el límite máximo de 245 caracteres (longitud actual: ' + clean.length + ').');
  }
  return clean;
}

/**
 * TÍTULO: Validación estricta de versión base entera no negativa
 * NOMBRE TÉCNICO: validateStrictIntegerVersion_
 * FECHA DE CREACIÓN: 27/11/2026
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * 
 * QUÉ HACE:
 *   Valida rigurosamente que baseVersion sea un entero válido no negativo.
 *   Rechaza terminantemente conversiones parciales (ej. '2abc' -> 2), decimales (ej. 2.5) o tipos inválidos.
 * 
 * PARA QUÉ SIRVE:
 *   Garantiza el control estricto de concurrencia optimista sin falsos positivos de versiones.
 * 
 * PARÁMETROS:
 *   @param {*} val Valor recibido en payload.baseVersion.
 * 
 * RESULTADO:
 *   @return {number} Entero validado.
 *   @throws {Error} Si no es un entero exacto.
 */
function validateStrictIntegerVersion_(val) {
  if (val === null || val === undefined) {
    throw new Error('CONFLICTO_VERSION: baseVersion es obligatorio para el control de concurrencia.');
  }
  if (typeof val === 'number') {
    if (!Number.isInteger(val) || val < 0) {
      throw new Error('CONFLICTO_VERSION: baseVersion debe ser un entero no negativo, no se aceptan números decimales: ' + val);
    }
    return val;
  }
  if (typeof val === 'string') {
    var trimmed = val.trim();
    if (!/^\d+$/.test(trimmed)) {
      throw new Error('CONFLICTO_VERSION: baseVersion debe ser una cadena entera exacta, no se aceptan conversiones parciales ni decimales: ' + val);
    }
    return parseInt(trimmed, 10);
  }
  throw new Error('CONFLICTO_VERSION: Tipo de baseVersion no admitido.');
}

/**
 * TÍTULO: Cálculo de hash de integridad en el servidor sobre todos los componentes
 * NOMBRE TÉCNICO: computeBundleIntegrityHash_
 * FECHA DE CREACIÓN: 27/11/2026
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * 
 * QUÉ HACE:
 *   Calcula en el servidor el hash SHA-256 incorporando todos los componentes del lote:
 *   ámbito escolar (ciclo, grupo, trimestre), calificaciones, desempeño, mensajes,
 *   configuración, actividades y banco de recomendaciones.
 *   NUNCA confía en hashes proporcionados por el cliente.
 * 
 * PARA QUÉ SIRVE:
 *   Garantiza la trazabilidad y la detección de alteraciones o reintentos inconsistentes.
 * 
 * PARÁMETROS:
 *   @param {Object} payload Datos del lote a sincronizar.
 * 
 * RESULTADO:
 *   @return {string} Hash SHA-256 en formato hexadecimal.
 */
function computeBundleIntegrityHash_(payload) {
  var raw = String(payload.cicloEscolar || '') + '|' +
            String(payload.grupo || '') + '|' +
            String(payload.trimestre || '') + '|' +
            JSON.stringify(payload.config || {}) + '|' +
            JSON.stringify(payload.actividades || []) + '|' +
            JSON.stringify(payload.calificaciones || []) + '|' +
            JSON.stringify(payload.desempeno || []) + '|' +
            JSON.stringify(payload.mensajes || []) + '|' +
            JSON.stringify(payload.bancoRecomendaciones || []);
  var bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, raw, Utilities.Charset.UTF_8);
  return bytes.map(function (b) { var hex = (b < 0 ? b + 256 : b).toString(16); return hex.length === 1 ? '0' + hex : hex; }).join('');
}

/**
 * TÍTULO: Consulta del paquete trimestral consolidado y banco de recomendaciones
 * NOMBRE TÉCNICO: getTrimesterEvaluationBundle
 * FECHA DE CREACIÓN: 27/11/2026
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * 
 * QUÉ HACE:
 *   Recupera las evaluaciones trimestrales de los alumnos pertenecientes EXCLUSIVAMENTE
 *   al último batchId sellado como 'consolidado' en EvalVersiones para el ámbito escolar.
 *   Recupera además el banco institucional de recomendaciones pedagógicas activas.
 *   Las filas huérfanas o 'en_preparacion' son completamente ignoradas.
 * 
 * PARA QUÉ SIRVE:
 *   Alimenta la vista de Evaluaciones Trimestrales con datos consistentes y auditados.
 * 
 * PARÁMETROS:
 *   @param {Object} payload { cicloEscolar: string, grupo: string, trimestre: number }
 *   @param {string} [sessionToken] Token docente validado con Zero-Trust.
 * 
 * RESULTADO:
 *   @return {Object} { ok: boolean, data: Object } con los registros consolidados y recomendaciones.
 */
function getTrimesterEvaluationBundle(payload, sessionToken) {
  var p = payload || {};
  var ciclo = String(p.cicloEscolar || '2026-2027').trim();
  var grupo = evalNormalizeGroup_(p.grupo || '4A');
  var t = evalValidateTrimester_(p.trimestre || 1);

  // Validación de seguridad Zero-Trust
  validateTeacherAuth_(sessionToken, grupo, ciclo);

  // 1. Obtener última versión consolidada en EvalVersiones aislada por ciclo, grupo y trimestre
  var versiones = (typeof readRows_ === 'function' ? readRows_('EvalVersiones') : []).filter(function (v) {
    return String(v.cicloEscolar) === ciclo &&
           evalNormalizeGroup_(v.grupo) === grupo &&
           parseInt(v.trimestre, 10) === t &&
           v.estadoConsolidacion === 'consolidado';
  });

  var latestVersionNum = 0;
  var latestBatchId = null;
  versiones.forEach(function (v) {
    var vNum = parseInt(v.versionNueva || v.version || 0, 10);
    if (vNum > latestVersionNum) {
      latestVersionNum = vNum;
      latestBatchId = v.batchId;
    }
  });

  // 2. Cargar Banco de Recomendaciones pedagógicas activas desde el servidor
  var bancoRecomendaciones = evalReadCommittedBank_();

  // 3. Cargar Configuración trimestral vinculada a la versión consolidada
  var configRows = (typeof readRows_ === 'function' ? readRows_('EvalConfig') : []).filter(function (c) {
    return String(c.cicloEscolar) === ciclo &&
           evalNormalizeGroup_(c.grupo) === grupo &&
           parseInt(c.trimestre, 10) === t &&
           (!latestBatchId || c.batchId === latestBatchId);
  });
  var configTrimestral = configRows.length > 0 ? configRows[configRows.length - 1] : null;

  // 4. Cargar Actividades seleccionadas de la versión consolidada
  var actividadesTrimestre = (typeof readRows_ === 'function' ? readRows_('EvalActividades') : []).filter(function (a) {
    return String(a.cicloEscolar) === ciclo &&
           evalNormalizeGroup_(a.grupo) === grupo &&
           parseInt(a.trimestre, 10) === t &&
           (!latestBatchId || a.batchId === latestBatchId);
  });

  // 5. Cargar Calificaciones consolidadas del lote vigente
  var calificacionesTrimestre = [];
  if (latestBatchId) {
    calificacionesTrimestre = (typeof readRows_ === 'function' ? readRows_('EvalCalificaciones') : []).filter(function (c) {
      return String(c.cicloEscolar) === ciclo &&
             evalNormalizeGroup_(c.grupo) === grupo &&
             parseInt(c.trimestre, 10) === t &&
             c.batchId === latestBatchId &&
             (c.estadoConsolidacion === 'consolidado' || !c.estadoConsolidacion);
    });
  }

  // 6. Cargar Desempeño Integral consolidado del lote vigente
  var desempenoTrimestre = [];
  if (latestBatchId) {
    desempenoTrimestre = (typeof readRows_ === 'function' ? readRows_('EvalDesempeno') : []).filter(function (d) {
      return String(d.cicloEscolar) === ciclo &&
             evalNormalizeGroup_(d.grupo) === grupo &&
             parseInt(d.trimestre, 10) === t &&
             d.batchId === latestBatchId &&
             (d.estadoConsolidacion === 'consolidado' || !d.estadoConsolidacion);
    });
  }

  // 7. Cargar Mensajes Familiares consolidados del lote vigente
  var mensajesTrimestre = [];
  if (latestBatchId) {
    mensajesTrimestre = (typeof readRows_ === 'function' ? readRows_('EvalMensajes') : []).filter(function (m) {
      return String(m.cicloEscolar) === ciclo &&
             evalNormalizeGroup_(m.grupo) === grupo &&
             parseInt(m.trimestre, 10) === t &&
             m.batchId === latestBatchId &&
             (m.estadoConsolidacion === 'consolidado' || !m.estadoConsolidacion);
    });
  }

  return {
    ok: true,
    data: {
      cicloEscolar: ciclo,
      grupo: grupo,
      trimestre: t,
      versionVigente: latestVersionNum,
      batchIdVigente: latestBatchId,
      config: configTrimestral,
      actividades: actividadesTrimestre,
      calificaciones: calificacionesTrimestre,
      desempeno: desempenoTrimestre,
      mensajes: mensajesTrimestre,
      bancoRecomendaciones: bancoRecomendaciones
    }
  };
}

/**
 * TÍTULO: Sincronización atómica por lotes con persistencia en dos fases y fusión de grupo
 * NOMBRE TÉCNICO: syncTrimesterEvaluationBatch
 * FECHA DE CREACIÓN: 27/11/2026
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * 
 * QUÉ HACE:
 *   Ejecuta la sincronización segura por lotes:
 *   1. Valida la sesión docente y extrae la identidad responsable del servidor.
 *   2. Adquiere bloqueo exclusivo mediante ScriptLock (evalWithScriptLock_).
 *   3. Exige coincidencia exacta de baseVersion entera (sin decimales ni conversiones parciales).
 *   4. Si es guardado parcial/individual, fusiona los datos con los alumnos consolidados previos.
 *   5. Escribe las filas con batchId propio en estado 'en_preparacion' (no sobrescribe V previas).
 *   6. Calcula el hash de integridad en el servidor abarcando todos los componentes modificados.
 *   7. Sella la nueva versión en EvalVersiones como 'consolidado' con snapshot compacto.
 * 
 * PARA QUÉ SIRVE:
 *   Garantiza persistencia atómica, trazabilidad, idempotencia y protección total contra pérdida de datos.
 * 
 * PARÁMETROS:
 *   @param {Object} payload Lote con batchId, baseVersion, calificaciones, etc.
 *   @param {string} sessionToken Token docente verificado.
 * 
 * RESULTADO:
 *   @return {Object} { ok: boolean, data: { batchId, version, ... } }
 */
function syncTrimesterEvaluationBatch(payload, sessionToken) {
  if (!payload || typeof payload !== 'object') {
    throw new Error('Payload inválido para sincronización de evaluación trimestral.');
  }

  var ciclo = String(payload.cicloEscolar || '2026-2027').trim();
  var grupo = evalNormalizeGroup_(payload.grupo || '4A');
  var t = evalValidateTrimester_(payload.trimestre || 1);
  var batchId = payload.batchId;

  if (!batchId) {
    throw new Error('Falta batchId único en el payload de sincronización.');
  }

  // 1. Autenticación Zero-Trust: la identidad responsable se obtiene DIRECTAMENTE de la sesión validada
  var authSession = validateTeacherAuth_(sessionToken, grupo, ciclo);
  var user = authSession.usuario; // NUNCA se confía en payload.usuario del cliente

  // 2. Concurrencia segura: Ejecutar bajo ScriptLock para compatibilidad con Web App
  return evalWithScriptLock_(function () {
    var now = (typeof now_ === 'function') ? now_() : new Date().toISOString();
    // Cálculo exclusivo de hash de integridad en el servidor (abarcando todos los componentes)
    var serverHash = computeBundleIntegrityHash_(payload);

    // Obtener padrón escolar autorizado del aula para validación estricta de pertenencia
    var authorizedStudentsMap = evalGetAuthorizedStudentsMap_(grupo);

    // 3. Verificar si este batchId ya fue consolidado previamente (Idempotencia)
    var versionesExistentes = (typeof readRows_ === 'function' ? readRows_('EvalVersiones') : []).filter(function (v) {
      return String(v.cicloEscolar) === ciclo &&
             evalNormalizeGroup_(v.grupo) === grupo &&
             parseInt(v.trimestre, 10) === t;
    });

    var matchingConsolidated = versionesExistentes.find(function (v) {
      return v.batchId === batchId && v.estadoConsolidacion === 'consolidado';
    });

    if (matchingConsolidated) {
      // Validar que el reintento tenga el mismo contenido y ámbito autorizado
      if (matchingConsolidated.payloadHash && matchingConsolidated.payloadHash !== serverHash) {
        throw new Error('CONFLICTO_REINTENTO_INCONSISTENTE: El batchId ya fue consolidado con un contenido diferente.');
      }
      return {
        ok: true,
        data: {
          batchId: batchId,
          version: matchingConsolidated.versionNueva || matchingConsolidated.version,
          idempotent: true,
          mensaje: 'Lote previamente consolidado.'
        }
      };
    }

    // 4. Control de versiones: Validación estricta de baseVersion y coincidencia EXACTA
    var clientBaseVer = validateStrictIntegerVersion_(payload.baseVersion);

    var latestConsolidatedVer = 0;
    var latestConsolidatedBatchId = null;
    versionesExistentes.forEach(function (v) {
      if (v.estadoConsolidacion === 'consolidado') {
        var vNum = parseInt(v.versionNueva || v.version || 0, 10);
        if (vNum > latestConsolidatedVer) {
          latestConsolidatedVer = vNum;
          latestConsolidatedBatchId = v.batchId;
        }
      }
    });

    if (clientBaseVer !== latestConsolidatedVer) {
      throw new Error('CONFLICTO_VERSION: La versión base (' + clientBaseVer +
                      ') no coincide exactamente con la versión vigente en el servidor (' + latestConsolidatedVer +
                      '). Debe recargar los datos antes de guardar.');
    }

    var newVersionNumber = latestConsolidatedVer + 1;

    // 5. Mecanismo de FUSIÓN (Merge) para capturas parciales o individuales:
    // Cargar registros de la versión consolidada anterior aislados por ciclo, grupo y trimestre
    var prevCalificacionesMap = {};
    var prevDesempenoMap = {};
    var prevMensajesMap = {};
    var prevConfig = null;
    var prevActividadesMap = {};

    if (latestConsolidatedBatchId) {
      var prevCalifs = (typeof readRows_ === 'function' ? readRows_('EvalCalificaciones') : []).filter(function (c) {
        return String(c.cicloEscolar) === ciclo &&
               evalNormalizeGroup_(c.grupo) === grupo &&
               parseInt(c.trimestre, 10) === t &&
               c.batchId === latestConsolidatedBatchId;
      });
      prevCalifs.forEach(function (c) {
        prevCalificacionesMap[c.alumnoId + '_' + c.campoFormativo] = c;
      });

      var prevDesemps = (typeof readRows_ === 'function' ? readRows_('EvalDesempeno') : []).filter(function (d) {
        return String(d.cicloEscolar) === ciclo &&
               evalNormalizeGroup_(d.grupo) === grupo &&
               parseInt(d.trimestre, 10) === t &&
               d.batchId === latestConsolidatedBatchId;
      });
      prevDesemps.forEach(function (d) {
        prevDesempenoMap[d.alumnoId] = d;
      });

      var prevMsgs = (typeof readRows_ === 'function' ? readRows_('EvalMensajes') : []).filter(function (m) {
        return String(m.cicloEscolar) === ciclo &&
               evalNormalizeGroup_(m.grupo) === grupo &&
               parseInt(m.trimestre, 10) === t &&
               m.batchId === latestConsolidatedBatchId;
      });
      prevMsgs.forEach(function (m) {
        prevMensajesMap[m.alumnoId] = m;
      });

      var prevConfigs = (typeof readRows_ === 'function' ? readRows_('EvalConfig') : []).filter(function (c) {
        return String(c.cicloEscolar) === ciclo &&
               evalNormalizeGroup_(c.grupo) === grupo &&
               parseInt(c.trimestre, 10) === t &&
               c.batchId === latestConsolidatedBatchId;
      });
      if (prevConfigs.length > 0) prevConfig = prevConfigs[prevConfigs.length - 1];

      var prevActs = (typeof readRows_ === 'function' ? readRows_('EvalActividades') : []).filter(function (a) {
        return String(a.cicloEscolar) === ciclo &&
               evalNormalizeGroup_(a.grupo) === grupo &&
               parseInt(a.trimestre, 10) === t &&
               a.batchId === latestConsolidatedBatchId;
      });
      prevActs.forEach(function (a) {
        prevActividadesMap[a.campoFormativo + '_' + a.criterio + '_' + a.actividadId] = a;
      });
    }

    var result = {
      batchId: batchId,
      version: newVersionNumber,
      calificacionesCount: 0,
      desempenoCount: 0,
      mensajesCount: 0
    };

    // 6. Preparación de Calificaciones (Fusión con alumnos existentes + nuevos cambios)
    var califBatchMap = {};
    for (var key in prevCalificacionesMap) {
      var prev = prevCalificacionesMap[key];
      var newId = ciclo + '_' + grupo + '_T' + t + '_' + batchId + '_' + prev.alumnoId + '_' + prev.campoFormativo;
      califBatchMap[key] = Object.assign({}, prev, {
        id: newId,
        batchId: batchId,
        estadoConsolidacion: 'en_preparacion',
        version: newVersionNumber,
        // Conservar autoría y fecha originales de registros que no hayan cambiado
        authorizedPor: prev.authorizedPor || '',
        authorizedAt: prev.authorizedAt || '',
        actualizadoEn: now,
        actualizadoPor: user
      });
    }

    if (Array.isArray(payload.calificaciones)) {
      payload.calificaciones.forEach(function (c) {
        if (!c.alumnoId || !c.campoFormativo) return;
        evalValidateStudentId_(c.alumnoId, authorizedStudentsMap, grupo);

        if (!EVAL_CAMPOS_FORMATIVOS[c.campoFormativo]) {
          throw new Error('Campo formativo no reconocido: ' + c.campoFormativo);
        }
        var califKey = c.alumnoId + '_' + c.campoFormativo;
        var prevRecord = prevCalificacionesMap[califKey];
        var califId = ciclo + '_' + grupo + '_T' + t + '_' + batchId + '_' + califKey;

        var authGrade = validateGradeValue_(c.authorizedGrade);
        var status = c.status || 'PENDIENTE';

        // Generar authorizedPor y authorizedAt desde el servidor cuando esté autorizada
        var authAt = '';
        var authPor = '';

        if (status === 'AUTORIZADA' || (authGrade !== null && authGrade !== undefined)) {
          // Si el registro previo ya estaba autorizado y no cambió de calificación/estado, conservar autoría y fecha originales
          if (prevRecord && prevRecord.status === 'AUTORIZADA' &&
              String(prevRecord.authorizedGrade) === String(authGrade) &&
              prevRecord.authorizedAt && prevRecord.authorizedPor) {
            authAt = prevRecord.authorizedAt;
            authPor = prevRecord.authorizedPor;
          } else {
            // Nueva autorización o cambio en calificación autorizada: generar en el servidor
            authAt = now;
            authPor = user;
          }
        }

        califBatchMap[califKey] = {
          id: califId,
          cicloEscolar: ciclo,
          grupo: grupo,
          trimestre: t,
          alumnoId: String(c.alumnoId),
          numeroLista: c.numeroLista ? parseInt(c.numeroLista, 10) : (authorizedStudentsMap[c.alumnoId] ? authorizedStudentsMap[c.alumnoId].numeroLista : null),
          campoFormativo: c.campoFormativo,
          modalidad: c.modalidad || 'asistida',
          desgloseAutoJson: JSON.stringify(c.desgloseAuto || {}),
          desgloseManualJson: JSON.stringify(c.desgloseManual || {}),
          examenGrade: validateGradeValue_(c.examenGrade),
          configAplicadaJson: JSON.stringify(c.configAplicada || {}),
          actividadesRefJson: JSON.stringify(c.actividadesRef || []),
          calcGradeOriginal: validateGradeValue_(c.calcGradeOriginal),
          calcGradePreliminar: validateGradeValue_(c.calcGradePreliminar),
          ajusteManual: validateGradeValue_(c.ajusteManual),
          motivoAjuste: c.motivoAjuste || '',
          authorizedGrade: authGrade,
          status: status,
          authorizedAt: authAt,
          authorizedPor: authPor,
          batchId: batchId,
          estadoConsolidacion: 'en_preparacion',
          version: newVersionNumber,
          actualizadoEn: now,
          actualizadoPor: user
        };
      });
    }

    var finalCalifBatch = Object.values(califBatchMap);
    result.calificacionesCount = finalCalifBatch.length;

    if (finalCalifBatch.length > 0) {
      if (typeof upsertObjectsBatch_ === 'function') {
        upsertObjectsBatch_('EvalCalificaciones', finalCalifBatch, 'id');
      } else if (typeof appendObjects_ === 'function') {
        appendObjects_('EvalCalificaciones', finalCalifBatch);
      }
    }

    // 7. Preparación de Desempeño Integral (Fusión por alumno e indicador)
    var desempBatchMap = {};
    for (var dKey in prevDesempenoMap) {
      var prevD = prevDesempenoMap[dKey];
      var newDId = ciclo + '_' + grupo + '_T' + t + '_' + batchId + '_' + prevD.alumnoId;
      desempBatchMap[dKey] = Object.assign({}, prevD, {
        id: newDId,
        batchId: batchId,
        estadoConsolidacion: 'en_preparacion',
        version: newVersionNumber,
        actualizadoEn: now,
        actualizadoPor: user
      });
    }

    if (Array.isArray(payload.desempeno)) {
      payload.desempeno.forEach(function (d) {
        if (!d.alumnoId) return;
        evalValidateStudentId_(d.alumnoId, authorizedStudentsMap, grupo);

        // Si ya existía un registro consolidado previo del mismo alumno, recuperar sus valoraciones previas
        var prevRatings = {};
        if (prevDesempenoMap[d.alumnoId] && prevDesempenoMap[d.alumnoId].valoracion) {
          try {
            prevRatings = JSON.parse(prevDesempenoMap[d.alumnoId].valoracion) || {};
          } catch (e) {
            prevRatings = {};
          }
        }

        var incomingRatings = d.ratings || {};
        // RECHAZAR indicadores y valoraciones inválidos en lugar de eliminarlos silenciosamente
        for (var indKey in incomingRatings) {
          if (!EVAL_INDICADORES_VALIDOS[indKey]) {
            throw new Error('INDICADOR_INVALIDO: El indicador "' + indKey + '" no existe en el catálogo oficial de 23 indicadores.');
          }
          var val = incomingRatings[indKey];
          if (!EVAL_NIVELES_VALIDOS[val]) {
            throw new Error('VALORACION_INVALIDA: La valoración "' + val + '" para el indicador "' + indKey + '" no es válida (esperado: E, B, R, A).');
          }
        }

        // Fusionar valoraciones: las valoraciones previas se conservan; las nuevas las actualizan/complementan
        var mergedRatings = Object.assign({}, prevRatings, incomingRatings);

        // Considerar revisado ÚNICAMENTE cuando exista una confirmación expresa Y estén completos los 23 indicadores válidos
        var explicitReviewed = (d.isReviewed === true || d.estadoRevision === 'revisado');
        var validIndicatorsCount = 0;
        for (var validKey in EVAL_INDICADORES_VALIDOS) {
          if (mergedRatings[validKey] && EVAL_NIVELES_VALIDOS[mergedRatings[validKey]]) {
            validIndicatorsCount++;
          }
        }
        var isFullyComplete = (validIndicatorsCount === 23);
        var finalEstadoRevision = (explicitReviewed && isFullyComplete) ? 'revisado' : 'pendiente';

        var desempId = ciclo + '_' + grupo + '_T' + t + '_' + batchId + '_' + d.alumnoId;
        desempBatchMap[d.alumnoId] = {
          id: desempId,
          cicloEscolar: ciclo,
          grupo: grupo,
          trimestre: t,
          alumnoId: String(d.alumnoId),
          numeroLista: d.numeroLista ? parseInt(d.numeroLista, 10) : (authorizedStudentsMap[d.alumnoId] ? authorizedStudentsMap[d.alumnoId].numeroLista : null),
          indicadorId: '23_INDICADORES_COMPLETOS',
          valoracion: JSON.stringify(mergedRatings),
          catalogoVersion: '2026_V1_23IND',
          estadoRevision: finalEstadoRevision,
          origen: d.source || 'manual',
          trimestreOrigen: t,
          batchId: batchId,
          estadoConsolidacion: 'en_preparacion',
          version: newVersionNumber,
          actualizadoEn: now,
          actualizadoPor: user
        };
      });
    }

    var finalDesempBatch = Object.values(desempBatchMap);
    result.desempenoCount = finalDesempBatch.length;

    if (finalDesempBatch.length > 0) {
      if (typeof upsertObjectsBatch_ === 'function') {
        upsertObjectsBatch_('EvalDesempeno', finalDesempBatch, 'id');
      } else if (typeof appendObjects_ === 'function') {
        appendObjects_('EvalDesempeno', finalDesempBatch);
      }
    }

    // 8. Preparación de Mensajes Familiares (Fusión con alumnos existentes)
    var msgBatchMap = {};
    for (var mKey in prevMensajesMap) {
      var prevM = prevMensajesMap[mKey];
      var newMId = ciclo + '_' + grupo + '_T' + t + '_' + batchId + '_' + prevM.alumnoId;
      msgBatchMap[mKey] = Object.assign({}, prevM, {
        id: newMId,
        batchId: batchId,
        estadoConsolidacion: 'en_preparacion',
        version: newVersionNumber,
        actualizadoEn: now,
        actualizadoPor: user
      });
    }

    if (Array.isArray(payload.mensajes)) {
      payload.mensajes.forEach(function (m) {
        if (!m.alumnoId) return;
        evalValidateStudentId_(m.alumnoId, authorizedStudentsMap, grupo);

        var cleanMsg = validateFamilyMessage_(m.mensaje);
        var msgId = ciclo + '_' + grupo + '_T' + t + '_' + batchId + '_' + m.alumnoId;
        msgBatchMap[m.alumnoId] = {
          id: msgId,
          cicloEscolar: ciclo,
          grupo: grupo,
          trimestre: t,
          alumnoId: String(m.alumnoId),
          numeroLista: m.numeroLista ? parseInt(m.numeroLista, 10) : (authorizedStudentsMap[m.alumnoId] ? authorizedStudentsMap[m.alumnoId].numeroLista : null),
          mensaje: cleanMsg,
          caracteres: cleanMsg.length,
          templateId: m.templateId || '',
          estadoRevision: m.estadoRevision || 'pendiente',
          batchId: batchId,
          estadoConsolidacion: 'en_preparacion',
          version: newVersionNumber,
          actualizadoEn: now,
          actualizadoPor: user
        };
      });
    }

    var finalMsgBatch = Object.values(msgBatchMap);
    result.mensajesCount = finalMsgBatch.length;

    if (finalMsgBatch.length > 0) {
      if (typeof upsertObjectsBatch_ === 'function') {
        upsertObjectsBatch_('EvalMensajes', finalMsgBatch, 'id');
      } else if (typeof appendObjects_ === 'function') {
        appendObjects_('EvalMensajes', finalMsgBatch);
      }
    }

    // 9. Persistencia de Configuración (con batchId para aislamiento total ante fallos)
    var activeConfigData = payload.config || prevConfig;
    if (activeConfigData && typeof activeConfigData === 'object') {
      var confId = ciclo + '_' + grupo + '_T' + t + '_' + batchId;
      var confRec = {
        id: confId,
        cicloEscolar: ciclo,
        grupo: grupo,
        trimestre: t,
        fechaInicio: activeConfigData.fechaInicio || '',
        fechaFin: activeConfigData.fechaFin || '',
        estadoPeriodo: activeConfigData.estadoPeriodo || 'activo',
        ponderacionesJson: JSON.stringify(activeConfigData.ponderaciones || {}),
        valoresTareasJson: JSON.stringify(activeConfigData.valoresTareas || {}),
        valoresConductaJson: JSON.stringify(activeConfigData.valoresConducta || {}),
        valoresParticipacionJson: JSON.stringify(activeConfigData.valoresParticipacion || {}),
        configAsistenciaJson: JSON.stringify(activeConfigData.configAsistencia || {}),
        ajustesAdminJson: JSON.stringify(activeConfigData.ajustesAdmin || {}),
        batchId: batchId,
        version: newVersionNumber,
        actualizadoEn: now,
        actualizadoPor: user
      };
      if (typeof upsertObjectsBatch_ === 'function') {
        upsertObjectsBatch_('EvalConfig', [confRec], 'id');
      }
    }

    // 10. Persistencia de Actividades Seleccionadas (con batchId para aislamiento)
    var actsToSave = [];
    if (Array.isArray(payload.actividades)) {
      actsToSave = payload.actividades;
    } else {
      actsToSave = Object.values(prevActividadesMap);
    }
    if (actsToSave.length > 0) {
      var actBatch = actsToSave.map(function (a) {
        var actId = ciclo + '_' + grupo + '_T' + t + '_' + batchId + '_' + a.campoFormativo + '_' + (a.criterio || 'general') + '_' + a.actividadId;
        return {
          id: actId,
          cicloEscolar: ciclo,
          grupo: grupo,
          trimestre: t,
          campoFormativo: a.campoFormativo,
          criterio: a.criterio || 'general',
          actividadId: String(a.actividadId),
          tipoActividad: a.tipoActividad || 'tarea',
          fechaActividad: a.fechaActividad || '',
          nombreActividad: a.nombreActividad || '',
          seleccionada: a.seleccionada !== false,
          peso: a.peso || 1.0,
          exclusionesAlumnosJson: JSON.stringify(a.exclusionesAlumnos || []),
          batchId: batchId,
          version: newVersionNumber,
          actualizadoEn: now,
          actualizadoPor: user
        };
      });
      if (typeof upsertObjectsBatch_ === 'function') {
        upsertObjectsBatch_('EvalActividades', actBatch, 'id');
      }
    }

    // 11. Preparar versiones del banco sin sobrescribir registros consolidados.
    if (Array.isArray(payload.bancoRecomendaciones)) {
      var bankVersionId = 'VER_' + ciclo + '_' + grupo + '_T' + t + '_' + batchId;
      var recBatch = evalPrepareBank_(payload.bancoRecomendaciones, bankVersionId, newVersionNumber, user, now);
      if (recBatch.length) upsertObjectsBatch_('EvalBancoRecomendaciones', recBatch, 'id');
    }
    SpreadsheetApp.flush();

    // 12. Consolidación atómica: Sellar el lote en EvalVersiones con snapshot compacto (<500 car)
    var versId = 'VER_' + ciclo + '_' + grupo + '_T' + t + '_' + batchId;
    var compactSnapshot = JSON.stringify({
      calificacionesCount: result.calificacionesCount,
      desempenoCount: result.desempenoCount,
      mensajesCount: result.mensajesCount,
      timestamp: now
    });

    var versionRec = {
      id: versId,
      cicloEscolar: ciclo,
      grupo: grupo,
      trimestre: t,
      alumnoId: 'GRUPO_COMPLETO',
      tipoOperacion: payload.tipoEvento || 'sincronizacion_evaluacion',
      batchId: batchId,
      versionAnterior: latestConsolidatedVer,
      versionNueva: newVersionNumber,
      motivoModificacion: payload.motivo || 'Sincronización consolidada de evaluación trimestral',
      snapshotJson: compactSnapshot,
      payloadHash: serverHash,
      creadoEn: now,
      usuario: user,
      estadoConsolidacion: 'consolidado'
    };

    if (typeof appendObjects_ === 'function') {
      appendObjects_('EvalVersiones', [versionRec]);
    } else if (typeof appendRow_ === 'function') {
      appendRow_('EvalVersiones', versionRec);
    }

    SpreadsheetApp.flush();
    return { ok: true, data: result };
  });
}

/**
 * TÍTULO: Migración idempotente de esquemas de evaluación
 * NOMBRE TÉCNICO: migrateEvaluacionesTrimestralesSchema_
 * FECHA DE CREACIÓN: 27/11/2026
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * 
 * QUÉ HACE:
 *   Crea exclusivamente las hojas faltantes del módulo respetando los esquemas oficiales.
 *   Si una hoja ya existe, valida que los encabezados coincidan sin tocar ni sobrescribir datos.
 * 
 * PARA QUÉ SIRVE:
 *   Garantiza que las 7 tablas existan con sus columnas oficiales sin riesgo de regresión.
 */
function migrateEvaluacionesTrimestralesSchema_(ss) {
  var targetSs = ss || (typeof getSpreadsheet_ === 'function' ? getSpreadsheet_() : SpreadsheetApp.getActiveSpreadsheet());
  if (!targetSs) throw new Error('No se pudo acceder al libro de Google Sheets.');

  var schema = typeof DB_EVALUACIONES !== 'undefined' ? DB_EVALUACIONES : {
    EvalConfig: [
      'id', 'cicloEscolar', 'grupo', 'trimestre', 'fechaInicio', 'fechaFin',
      'estadoPeriodo', 'ponderacionesJson', 'valoresTareasJson', 'valoresConductaJson',
      'valoresParticipacionJson', 'configAsistenciaJson', 'ajustesAdminJson', 'batchId', 'version',
      'actualizadoEn', 'actualizadoPor'
    ],
    EvalActividades: [
      'id', 'cicloEscolar', 'grupo', 'trimestre', 'campoFormativo', 'criterio',
      'actividadId', 'tipoActividad', 'fechaActividad', 'nombreActividad',
      'seleccionada', 'peso', 'exclusionesAlumnosJson', 'batchId', 'version', 'actualizadoEn', 'actualizadoPor'
    ],
    EvalCalificaciones: [
      'id', 'cicloEscolar', 'grupo', 'trimestre', 'alumnoId', 'numeroLista',
      'campoFormativo', 'modalidad', 'desgloseAutoJson', 'desgloseManualJson', 'examenGrade',
      'configAplicadaJson', 'actividadesRefJson', 'calcGradeOriginal', 'calcGradePreliminar',
      'ajusteManual', 'motivoAjuste', 'authorizedGrade', 'status', 'authorizedAt', 'authorizedPor',
      'batchId', 'estadoConsolidacion', 'version', 'actualizadoEn', 'actualizadoPor'
    ],
    EvalDesempeno: [
      'id', 'cicloEscolar', 'grupo', 'trimestre', 'alumnoId', 'numeroLista',
      'indicadorId', 'valoracion', 'catalogoVersion', 'estadoRevision', 'origen',
      'trimestreOrigen', 'batchId', 'estadoConsolidacion', 'version', 'actualizadoEn', 'actualizadoPor'
    ],
    EvalMensajes: [
      'id', 'cicloEscolar', 'grupo', 'trimestre', 'alumnoId', 'numeroLista',
      'mensaje', 'caracteres', 'templateId', 'estadoRevision', 'batchId', 'estadoConsolidacion',
      'version', 'actualizadoEn', 'actualizadoPor'
    ],
    EvalBancoRecomendaciones: [
      'id', 'categoria', 'campoFormativo', 'nivel', 'texto', 'caracteres',
      'estadoAprobacion', 'activo', 'creadoEn', 'actualizadoEn', 'actualizadoPor', 'version', 'recomendacionId', 'loteVersionId'
    ],
    EvalVersiones: [
      'id', 'cicloEscolar', 'grupo', 'trimestre', 'alumnoId', 'tipoOperacion',
      'batchId', 'versionAnterior', 'versionNueva', 'motivoModificacion', 'snapshotJson',
      'payloadHash', 'creadoEn', 'usuario', 'estadoConsolidacion'
    ]
  };

  var created = [];
  var verified = [];

  for (var sheetName in schema) {
    var expectedHeaders = schema[sheetName];
    var sheet = targetSs.getSheetByName(sheetName);

    if (!sheet) {
      sheet = targetSs.insertSheet(sheetName);
      sheet.getRange(1, 1, 1, expectedHeaders.length).setValues([expectedHeaders]);
      sheet.getRange(1, 1, 1, expectedHeaders.length).setFontWeight('bold');
      sheet.setFrozenRows(1);
      created.push(sheetName);
    } else {
      var lastCol = sheet.getLastColumn();
      if (lastCol === 0) {
        sheet.getRange(1, 1, 1, expectedHeaders.length).setValues([expectedHeaders]);
        sheet.getRange(1, 1, 1, expectedHeaders.length).setFontWeight('bold');
        sheet.setFrozenRows(1);
        created.push(sheetName + ' (encabezados inicializados)');
      } else {
        var currentHeaders = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
        for (var i = 0; i < expectedHeaders.length; i++) {
          if (i >= currentHeaders.length || String(currentHeaders[i] || '').trim() !== expectedHeaders[i]) {
            throw new Error('Esquema incompatible en hoja "' + sheetName + '" en columna ' + (i + 1) + 
                            ': esperado "' + expectedHeaders[i] + '", encontrado "' + (currentHeaders[i] || '') + '".');
          }
        }
        verified.push(sheetName);
      }
    }
  }

  return { ok: true, created: created, verified: verified };
}

/**
 * TÍTULO: Verificación previa de instalación (SOLO LECTURA — 0 ESCRITURAS)
 * NOMBRE TÉCNICO: verificarInstalacionEvaluacionesTrimestrales
 * FECHA DE CREACIÓN: 27/11/2026
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * 
 * QUÉ HACE:
 *   Inspecciona la existencia de las 7 hojas de evaluaciones trimestrales en el libro de cálculo.
 *   NO ejecuta migraciones, NO invoca setupSystem_, NO modifica esquemas y realiza CERO (0) escrituras.
 * 
 * PARA QUÉ SIRVE:
 *   Permite al docente verificar el estado actual antes de autorizar la creación física de hojas.
 * 
 * RESULTADO:
 *   @return {Object} Reporte JSON con hojasExistentes, hojasPorCrear, escrituras: 0.
 */
function verificarInstalacionEvaluacionesTrimestrales() {
  var ss = typeof getSpreadsheet_ === 'function' ? getSpreadsheet_() : SpreadsheetApp.getActiveSpreadsheet();
  if (!ss) throw new Error('No se pudo acceder al libro de cálculo.');

  var schema = typeof DB_EVALUACIONES !== 'undefined' ? DB_EVALUACIONES : null;
  var targetSheets = schema ? Object.keys(schema) : [
    'EvalConfig', 'EvalActividades', 'EvalCalificaciones',
    'EvalDesempeno', 'EvalMensajes', 'EvalBancoRecomendaciones', 'EvalVersiones'
  ];

  var existentes = [];
  var faltantes = [];
  var erroresEsquema = [];
  var encabezadosCompatibles = true;

  targetSheets.forEach(function (name) {
    var sheet = ss.getSheetByName(name);
    if (sheet) {
      existentes.push(name);
      // Validar encabezados exactos si la hoja ya existe físicamente
      if (schema && schema[name]) {
        var expectedHeaders = schema[name];
        var lastCol = sheet.getLastColumn();
        if (lastCol < expectedHeaders.length) {
          erroresEsquema.push({
            hoja: name,
            error: 'Columnas insuficientes: esperado ' + expectedHeaders.length + ', encontrado ' + lastCol
          });
          encabezadosCompatibles = false;
        } else {
          var actualHeaders = sheet.getRange(1, 1, 1, expectedHeaders.length).getValues()[0] || [];
          for (var i = 0; i < expectedHeaders.length; i++) {
            if (String(actualHeaders[i] || '').trim() !== String(expectedHeaders[i]).trim()) {
              erroresEsquema.push({
                hoja: name,
                columna: i + 1,
                esperado: expectedHeaders[i],
                encontrado: actualHeaders[i]
              });
              encabezadosCompatibles = false;
            }
          }
        }
      }
    } else {
      faltantes.push(name);
    }
  });

  // Validar dependencias necesarias del entorno
  var dependenciasFaltantes = [];
  if (typeof SpreadsheetApp === 'undefined') dependenciasFaltantes.push('SpreadsheetApp');
  if (typeof LockService === 'undefined') dependenciasFaltantes.push('LockService');
  if (typeof Utilities === 'undefined') dependenciasFaltantes.push('Utilities');
  if (typeof DB_EVALUACIONES === 'undefined' && typeof DB === 'undefined') dependenciasFaltantes.push('DB_EVALUACIONES');

  // Validar disponibilidad del validador de sesión auténtico
  var validadorDisponible = (typeof validarSesion_ === 'function');

  // Comprobar si existen bloqueos pendientes
  var bloqueoPendiente = false;
  if (typeof LockService !== 'undefined' && LockService.getScriptLock) {
    var testLock = LockService.getScriptLock();
    var acquired = false;
    try {
      acquired = testLock.tryLock(500);
      if (!acquired) {
        bloqueoPendiente = true;
      }
    } catch (e) {
      bloqueoPendiente = true;
    } finally {
      if (acquired) {
        try { testLock.releaseLock(); } catch (e) {}
      }
    }
  }

  // NO devolver listoParaCrearHojas: true de manera incondicional
  var listoParaCrearHojas = (
    dependenciasFaltantes.length === 0 &&
    validadorDisponible === true &&
    bloqueoPendiente === false &&
    encabezadosCompatibles === true
  );

  var r = {
    ok: listoParaCrearHojas,
    totalHojasRequeridas: targetSheets.length,
    hojasExistentes: existentes,
    hojasPorCrear: faltantes,
    erroresEsquema: erroresEsquema,
    encabezadosCompatibles: encabezadosCompatibles,
    dependenciasFaltantes: dependenciasFaltantes,
    validadorDisponible: validadorDisponible,
    bloqueoPendiente: bloqueoPendiente,
    escrituras: 0,
    listoParaCrearHojas: listoParaCrearHojas
  };

  console.log(JSON.stringify(r));
  return r;
}

/**
 * TÍTULO: Aplicación controlada de la instalación de hojas
 * NOMBRE TÉCNICO: aplicarInstalacionEvaluacionesTrimestrales
 * FECHA DE CREACIÓN: 27/11/2026
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * 
 * QUÉ HACE:
 *   Crea exclusivamente las hojas faltantes del módulo bajo ScriptLock (evalWithScriptLock_).
 *   Requiere autorización explícita previa del docente.
 * 
 * PARA QUÉ SIRVE:
 *   Inicializa físicamente las tablas requeridas sin alterar ninguna hoja existente.
 */
function aplicarInstalacionEvaluacionesTrimestrales() {
  return evalWithScriptLock_(function () {
    var mig = migrateEvaluacionesTrimestralesSchema_();
    SpreadsheetApp.flush();
    var ver = verificarInstalacionEvaluacionesTrimestrales();
    if (ver.hojasPorCrear.length > 0) {
      throw new Error('Verificación posterior fallida: aún quedan hojas por inicializar.');
    }
    var r = {
      ok: true,
      mensaje: 'Instalación de hojas completada con éxito.',
      hojasCreadas: mig.created,
      hojasVerificadas: mig.verified
    };
    console.log(JSON.stringify(r));
    return r;
  }, 20000);
}

/**
 * TÍTULO: Lectura del banco consolidado
 * FECHA DE MODIFICACIÓN: 27/11/2026
 * PROPÓSITO Y MOTIVO: Devuelve la última revisión por ID lógico según el orden de sellado; ignora escrituras parciales.
 * HISTORIAL: 27/11/2026 — integración focalizada 7A.3.
 */
function evalReadCommittedBank_() {
  var order = new Map();
  readRows_('EvalVersiones').forEach(function(v,i) { if (v.estadoConsolidacion === 'consolidado') order.set(String(v.id),i); });
  var latest = new Map();
  readRows_('EvalBancoRecomendaciones').forEach(function(r) {
    if (!r.loteVersionId || !order.has(String(r.loteVersionId))) return;
    var id = String(r.recomendacionId), rank = order.get(String(r.loteVersionId));
    if (!latest.has(id) || latest.get(id).rank <= rank) latest.set(id,{rank:rank,row:r});
  });
  return Array.from(latest.values()).map(function(x) { return Object.assign({},x.row,{id:x.row.recomendacionId}); });
}
/**
 * TÍTULO: Preparación de revisiones del banco
 * FECHA DE MODIFICACIÓN: 27/11/2026
 * PROPÓSITO Y MOTIVO: Valida texto y estado, genera ID físico por lote; no altera versiones anteriores.
 * HISTORIAL: 27/11/2026 — integración focalizada 7A.3.
 */
function evalPrepareBank_(rows, versionId, version, user, now) {
  var seen = new Set();
  return rows.map(function(r,i) {
    var id = String(r.id || ('REC_' + versionId + '_' + i));
    if (seen.has(id)) throw new Error('Recomendación duplicada.');
    seen.add(id);
    var text = String(r.texto || '').trim();
    var status = r.estadoAprobacion || 'pendiente';
    if (!text || text.length > 245 || !['pendiente','aprobada','rechazada'].includes(status)) throw new Error('Recomendación inválida.');
    return {id:JSON.stringify([versionId,id]),recomendacionId:id,loteVersionId:versionId,categoria:r.categoria || 'pedagogica',campoFormativo:r.campoFormativo || '',nivel:r.nivel || 'E',texto:text,caracteres:text.length,estadoAprobacion:status,activo:r.activo !== false,creadoEn:r.creadoEn || now,actualizadoEn:now,actualizadoPor:user,version:version};
  });
}
