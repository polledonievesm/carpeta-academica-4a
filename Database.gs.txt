 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * QUÉ HACE: Crea las hojas esenciales del sistema, migra alumnos antiguos y carga datos semilla.

 * PARÁMETROS: Ninguno.

 * HISTORIAL:
 *   - Fecha no registrada: Creación inicial.
 *   - 27/11/2026 (Modificación documental): Documentación obligatoria JSDoc.
 */
function setupSystem_() {
  const ss = getSpreadsheet_();
  migrateLegacyStudents_(ss);
  Object.keys(DB).forEach(name => ensureSheet_(ss, name, DB[name]));
  seedConfig_();
  seedSubjects_();
  seedRubrics_();
  seedObservationBankV2_();
  seedEvaluationPeriods_();
  configurar_();
  CacheService.getScriptCache().put('CARPETA_SCHEMA_READY_V153', '1', 600);
  CacheService.getScriptCache().put('CARPETA_DIRECTORY_READY_V153', '1', 600);
  return { ok: true, message: 'Sistema instalado correctamente.', spreadsheetId: ss.getId() };
}

/**
 * TÍTULO: Migración de encabezados legados de alumnos
 * NOMBRE TÉCNICO: migrateLegacyStudents_
 * FECHA DE CREACIÓN: Fecha de creación no registrada
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * QUÉ HACE: Convierte la estructura de columnas antigua de la hoja Alumnos al formato estándar.
 * PARA QUÉ SIRVE: Asegura compatibilidad con libros escolares previos.
 * POR QUÉ SE CREÓ: Para actualizar esquemas antiguos sin pérdida de información de alumnos.
 * PARÁMETROS: @param {GoogleAppsScript.Spreadsheet.Spreadsheet} ss Libro activo.
 * RESULTADO: {void}
 * DEPENDENCIAS: SpreadsheetApp, DB.Alumnos.
 * HISTORIAL:
 *   - Fecha no registrada: Creación inicial.
 *   - 27/11/2026 (Modificación documental): Documentación obligatoria JSDoc.
 */
function migrateLegacyStudents_(ss) {
  const sheet = ss.getSheetByName('Alumnos');
  if (!sheet || sheet.getLastRow() < 1) return;
  const headers = sheet.getRange(1, 1, 1, Math.max(sheet.getLastColumn(), 1)).getDisplayValues()[0];
  if (headers.indexOf('nombreCompleto') >= 0) return;
  const legacyId = headers.indexOf('ID');
  const legacyName = headers.indexOf('Nombre completo');
  const legacyActive = headers.indexOf('Activo');
  if (legacyId < 0 || legacyName < 0) return;

  const backupName = 'Alumnos respaldo antes de unificar';
  if (!ss.getSheetByName(backupName)) sheet.copyTo(ss).setName(backupName);
  const source = sheet.getLastRow() > 1
    ? sheet.getRange(2, 1, sheet.getLastRow() - 1, sheet.getLastColumn()).getValues()
    : [];
  const now = now_();
  const migrated = source.filter(row => row[legacyId] && row[legacyName]).map(row => ({
    id: String(row[legacyId]),
    nombreCompleto: String(row[legacyName]),
    apellidoPaterno: '', apellidoMaterno: '', nombres: '', fechaNacimiento: '', edad: '',
    tutorNombre: '', tutorParentesco: '', tutorTelefono: '', tutorUsuario: '',
    activo: legacyActive < 0 || row[legacyActive] !== false,
    creadoEn: now, actualizadoEn: now
  }));
  sheet.clear();
  sheet.getRange(1, 1, 1, DB.Alumnos.length).setValues([DB.Alumnos]);
  if (migrated.length) {
    const rows = migrated.map(obj => DB.Alumnos.map(h => obj[h] === undefined ? '' : obj[h]));
    sheet.getRange(2, 1, rows.length, DB.Alumnos.length).setValues(rows);
  }
  sheet.setFrozenRows(1);
  sheet.getRange(1, 1, 1, DB.Alumnos.length).setFontWeight('bold').setBackground('#0B315D').setFontColor('#FFFFFF');
}

/**
 * TÍTULO: Verificación y creación idempotente de hoja con encabezados
 * NOMBRE TÉCNICO: ensureSheet_
 * FECHA DE CREACIÓN: Fecha de creación no registrada
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * QUÉ HACE: Crea una hoja si no existe y garantiza que la primera fila contenga los encabezados.
 * PARA QUÉ SIRVE: Garantiza que la tabla física exista y tenga la estructura de columnas requerida.
 * POR QUÉ SE CREÓ: Para soporte idempotente de esquemas relacionales.
 * PARÁMETROS:
 *   @param {GoogleAppsScript.Spreadsheet.Spreadsheet} ss Libro.
 *   @param {string} name Nombre de la tabla/hoja.
 *   @param {Array<string>} headers Lista de columnas oficiales.
 * RESULTADO: {GoogleAppsScript.Spreadsheet.Sheet} Hoja verificada.
 * DEPENDENCIAS: SpreadsheetApp.
 * HISTORIAL:
 *   - Fecha no registrada: Creación inicial.
 *   - 27/11/2026 (Modificación documental): Documentación obligatoria JSDoc.
 */
function ensureSheet_(ss, name, headers) {
  let sheet = ss.getSheetByName(name);
  if (!sheet) sheet = ss.insertSheet(name);
  if (sheet.getLastRow() === 0) {
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    sheet.setFrozenRows(1);
    sheet.getRange(1, 1, 1, headers.length)
      .setFontWeight('bold')
      .setBackground('#0B315D')
      .setFontColor('#FFFFFF');
    sheet.autoResizeColumns(1, headers.length);
  } else {
    const current = sheet.getRange(1, 1, 1, Math.max(sheet.getLastColumn(), 1)).getDisplayValues()[0].map(String);
    if (name === 'Alumnos') {
      const expected = DB.Alumnos;
      if (current.length !== expected.length || current.some(function (h, i) { return h !== expected[i]; })) {
        throw new Error('La hoja Alumnos requiere revisión del esquema. Ejecuta únicamente la migración puntual autorizada antes de continuar; no se agregaron columnas automáticamente.');
      }
    }
    const missing = headers.filter(function (header) { return current.indexOf(header) < 0; });
    if (missing.length) {
      const start = current.length + 1;
      sheet.getRange(1, start, 1, missing.length).setValues([missing]);
      sheet.getRange(1, start, 1, missing.length).setFontWeight('bold').setBackground('#0B315D').setFontColor('#FFFFFF');
      sheet.autoResizeColumns(start, missing.length);
    }
  }
  return sheet;
}

/**
 * TÍTULO: Lectura estructurada de filas como objetos JSON
 * NOMBRE TÉCNICO: readRows_
 * FECHA DE CREACIÓN: Fecha de creación no registrada
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * QUÉ HACE: Lee el rango de datos de una hoja y mapea cada fila a un objeto según sus encabezados.
 * PARA QUÉ SIRVE: Es la función universal de lectura de datos del sistema.
 * POR QUÉ SE CREÓ: Para abstraer la manipulación de filas y columnas de Sheets a objetos JavaScript.
 * PARÁMETROS: @param {string} sheetName Nombre de la hoja/tabla a consultar.
 * RESULTADO: {Array<Object>} Lista de objetos con los registros de la tabla.
 * DEPENDENCIAS: getSpreadsheet_, serializeCell_.
 * HISTORIAL:
 *   - Fecha no registrada: Creación inicial.
 *   - 27/11/2026 (Modificación documental): Documentación obligatoria JSDoc.
 */
function readRows_(sheetName) {
  const sheet = getSpreadsheet_().getSheetByName(sheetName);
  if (!sheet || sheet.getLastRow() < 2) return [];
  const values = sheet.getDataRange().getValues();
  const headers = values.shift().map(String);
  return values.filter(row => row.some(v => v !== '')).map(row => {
    const obj = {};
   headers.forEach((h, i) => obj[h] = serializeCell_(row[i], h));
    return obj;
  });
}

/**
 * TÍTULO: Inserción de múltiples objetos como filas en una tabla
 * NOMBRE TÉCNICO: appendObjects_
 * FECHA DE CREACIÓN: Fecha de creación no registrada
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * QUÉ HACE: Transforma un array de objetos a matrices de celdas y los añade al final de la hoja.
 * PARA QUÉ SIRVE: Escritura por lotes eficiente de nuevos registros.
 * POR QUÉ SE CREÓ: Para evitar llamadas individuales de appendRow que degradan el rendimiento.
 * PARÁMETROS:
 *   @param {string} sheetName Nombre de la hoja.
 *   @param {Array<Object>} objects Lista de registros a insertar.
 * RESULTADO: {void}
 * DEPENDENCIAS: getSpreadsheet_, DB, DB_EVALUACIONES.
 * HISTORIAL:
 *   - Fecha no registrada: Creación inicial.
 *   - 27/11/2026 (Modificación funcional y documental): Soporte para DB_EVALUACIONES y JSDoc.
 */
function appendObjects_(sheetName, objects) {
  if (!objects.length) return;
  const sheet = getSpreadsheet_().getSheetByName(sheetName);
  const headers = DB[sheetName] || (typeof DB_EVALUACIONES !== 'undefined' ? DB_EVALUACIONES[sheetName] : null);
  const rows = objects.map(obj => headers.map(h => obj[h] === undefined ? '' : obj[h]));
  sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, headers.length).setValues(rows);
}

/**
 * TÍTULO: Actualización o inserción individual de un registro por clave
 * NOMBRE TÉCNICO: upsertObject_
 * FECHA DE CREACIÓN: Fecha de creación no registrada
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * QUÉ HACE: Busca un registro por su campo clave; si existe lo actualiza, de lo contrario lo inserta.
 * PARA QUÉ SIRVE: Garantiza que no se generen duplicados al guardar una entidad.
 * POR QUÉ SE CREÓ: Operación atómica de actualización sobre registros individuales.
 * PARÁMETROS:
 *   @param {string} sheetName Nombre de la hoja.
 *   @param {Object} object Registro a guardar.
 *   @param {string} idField Nombre de la columna identificadora.
 * RESULTADO: {void}
 * DEPENDENCIAS: getSpreadsheet_, DB, DB_EVALUACIONES.
 * HISTORIAL:
 *   - Fecha no registrada: Creación inicial.
 *   - 27/11/2026 (Modificación funcional y documental): Soporte para DB_EVALUACIONES y JSDoc.
 */
function upsertObject_(sheetName, object, idField) {
  const sheet = getSpreadsheet_().getSheetByName(sheetName);
  const headers = DB[sheetName] || (typeof DB_EVALUACIONES !== 'undefined' ? DB_EVALUACIONES[sheetName] : null);
  const idColumn = headers.indexOf(idField) + 1;
  const ids = sheet.getLastRow() > 1
    ? sheet.getRange(2, idColumn, sheet.getLastRow() - 1, 1).getDisplayValues().flat()
    : [];
  const index = ids.indexOf(String(object[idField]));
  const row = headers.map(h => object[h] === undefined ? '' : object[h]);
  if (index === -1) sheet.appendRow(row);
  else sheet.getRange(index + 2, 1, 1, headers.length).setValues([row]);
}

/**
 * TÍTULO: Actualización o inserción masiva y optimizada por lotes
 * NOMBRE TÉCNICO: upsertObjectsBatch_
 * FECHA DE CREACIÓN: Fecha de creación no registrada
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * QUÉ HACE: Agrupa registros existentes en rangos contiguos para actualizarlos en un número mínimo
 *           de llamadas a setValues(), e inserta las nuevas filas al final.
 * PARA QUÉ SIRVE: Operación de persistencia por lotes de alto rendimiento en Google Sheets.
 * POR QUÉ SE CREÓ: Para resolver guardados masivos (ej. 35 alumnos) sin agotar cuotas ni timeouts.
 * PARÁMETROS:
 *   @param {string} sheetName Nombre de la hoja.
 *   @param {Array<Object>} objects Lista de registros.
 *   @param {string} idField Columna identificadora.
 * RESULTADO: {void}
 * DEPENDENCIAS: getSpreadsheet_, DB, DB_EVALUACIONES.
 * HISTORIAL:
 *   - Fecha no registrada: Creación inicial.
 *   - 27/11/2026 (Modificación funcional y documental): Soporte para DB_EVALUACIONES y JSDoc.
 */
function upsertObjectsBatch_(sheetName, objects, idField) {
  if (!objects || !objects.length) return;
  const sheet = getSpreadsheet_().getSheetByName(sheetName);
  const headers = DB[sheetName] || (typeof DB_EVALUACIONES !== 'undefined' ? DB_EVALUACIONES[sheetName] : null);
  if (!sheet || !headers) throw new Error('No se encontró la tabla ' + sheetName + '.');
  const idColumn = headers.indexOf(idField) + 1;
  const ids = sheet.getLastRow() > 1 ? sheet.getRange(2, idColumn, sheet.getLastRow() - 1, 1).getDisplayValues().flat() : [];
  const updates = [], additions = [];
  objects.forEach(function (object) {
    const index = ids.indexOf(String(object[idField]));
    const values = headers.map(function (header) { return object[header] === undefined ? '' : object[header]; });
    if (index < 0) additions.push(values); else updates.push({ row: index + 2, values: values });
  });
  updates.sort(function (a, b) { return a.row - b.row; });
  let group = [];
  const flush = function () {
    if (!group.length) return;
    sheet.getRange(group[0].row, 1, group.length, headers.length).setValues(group.map(function (item) { return item.values; }));
    group = [];
  };
  updates.forEach(function (item) {
    if (group.length && item.row !== group[group.length - 1].row + 1) flush();
    group.push(item);
  });
  flush();
  if (additions.length) sheet.getRange(sheet.getLastRow() + 1, 1, additions.length, headers.length).setValues(additions);
}

/**
 * TÍTULO: Eliminación lógica o vaciado de registros por lista de IDs
 * NOMBRE TÉCNICO: clearObjectsByIdsBatch_
 * FECHA DE CREACIÓN: Fecha de creación no registrada
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * QUÉ HACE: Localiza y limpia los valores de filas correspondientes a identificadores dados.
 * PARA QUÉ SIRVE: Eliminación en bloque de registros relacionados.
 * POR QUÉ SE CREÓ: Optimización de bajas y limpieza de relaciones dependientes.
 * PARÁMETROS:
 *   @param {string} sheetName Nombre de la tabla.
 *   @param {Array<string>} idsToClear Lista de IDs a limpiar.
 *   @param {string} idField Nombre del campo identificador.
 * RESULTADO: {void}
 * DEPENDENCIAS: getSpreadsheet_, DB.
 * HISTORIAL:
 *   - Fecha no registrada: Creación inicial.
 *   - 27/11/2026 (Modificación documental): Documentación obligatoria JSDoc.
 */
function clearObjectsByIdsBatch_(sheetName, idsToClear, idField) {
  const wanted = new Set((idsToClear || []).map(String));
  if (!wanted.size) return;
  const sheet = getSpreadsheet_().getSheetByName(sheetName), headers = DB[sheetName];
  if (!sheet || sheet.getLastRow() < 2) return;
  const idColumn = headers.indexOf(idField) + 1;
  const ids = sheet.getRange(2, idColumn, sheet.getLastRow() - 1, 1).getDisplayValues().flat();
  const rows = ids.map(function (id, index) { return wanted.has(String(id)) ? index + 2 : 0; }).filter(Boolean).sort(function (a, b) { return a - b; });
  let start = 0, previous = 0;
  rows.concat([0]).forEach(function (row) {
    if (start && row !== previous + 1) { sheet.getRange(start, 1, previous - start + 1, headers.length).clearContent(); start = 0; }
    if (row) { if (!start) start = row; previous = row; }
  });
}

/**
 * TÍTULO: Reemplazo integral del contenido de una tabla
 * NOMBRE TÉCNICO: replaceObjects_
 * FECHA DE CREACIÓN: Fecha de creación no registrada
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * QUÉ HACE: Limpia las filas de datos preservando la fila 1 de encabezados e inserta la nueva lista.
 * PARA QUÉ SIRVE: Sobreescritura completa de tablas de configuración o catálogos sincronizados.
 * POR QUÉ SE CREÓ: Para reordenamientos completos o restauraciones.
 * PARÁMETROS:
 *   @param {string} sheetName Nombre de la hoja.
 *   @param {Array<Object>} objects Nuevos registros.
 * RESULTADO: {void}
 * DEPENDENCIAS: getSpreadsheet_, DB.
 * HISTORIAL:
 *   - Fecha no registrada: Creación inicial.
 *   - 27/11/2026 (Modificación documental): Documentación obligatoria JSDoc.
 */
function replaceObjects_(sheetName, objects) {
  const sheet = getSpreadsheet_().getSheetByName(sheetName);
  const headers = DB[sheetName] || (typeof DB_EVALUACIONES !== 'undefined' ? DB_EVALUACIONES[sheetName] : null);
  if (!sheet || !headers) throw new Error('No se encontró la tabla ' + sheetName + '.');
  const missingRows = objects.length + 1 - sheet.getMaxRows();
  if (missingRows > 0) sheet.insertRowsAfter(sheet.getMaxRows(), missingRows);
  const rowsToClear = Math.max(sheet.getLastRow() - 1, objects.length);
  if (rowsToClear > 0) sheet.getRange(2, 1, rowsToClear, headers.length).clearContent();
  if (objects.length) {
    sheet.getRange(2, 1, objects.length, headers.length)
      .setValues(objects.map(obj => headers.map(h => obj[h] === undefined ? '' : obj[h])));
  }
}

/**
 * TÍTULO: Serialización y formateo de celda a tipo nativo JavaScript
 * NOMBRE TÉCNICO: serializeCell_
 * FECHA DE CREACIÓN: Fecha de creación no registrada
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * QUÉ HACE: Normaliza fechas, horas, números y booleanos devueltos por Google Sheets.
 * PARA QUÉ SIRVE: Evita inconsistencias de formato en la capa de datos de la API.
 * POR QUÉ SE CREÓ: Para estandarizar los tipos de datos devueltos a la aplicación web.
 * PARÁMETROS:
 *   @param {*} value Valor crudo de la celda.
 *   @param {string} header Nombre de la columna.
 * RESULTADO: {*} Valor normalizado (string ISO, número, booleano).
 * DEPENDENCIAS: Utilities, Session.getScriptTimeZone.
 * HISTORIAL:
 *   - Fecha no registrada: Creación inicial.
 *   - 27/11/2026 (Modificación documental): Documentación obligatoria JSDoc.
 */
function serializeCell_(value, header) {
  if (value instanceof Date) {
    if (header === 'horaInicio' || header === 'horaFin') {
      return Utilities.formatDate(
        value, Session.getScriptTimeZone(), 'HH:mm'
      );
    }
    return Utilities.formatDate(
      value, Session.getScriptTimeZone(), 'yyyy-MM-dd'
    );
  }
  return value;
}

 */
/**
 * TÍTULO: Migración estructural del esquema de Registro Diario
 * NOMBRE TÉCNICO: migrateDailyDatabaseSchema_
 * FECHA DE CREACIÓN: Fecha de creación no registrada
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * QUÉ HACE: Inspecciona y agrega columnas faltantes en las hojas Clases y Evaluaciones.
 * PARA QUÉ SIRVE: Mantiene actualizadas las tablas del registro diario a sus columnas oficiales.
 * POR QUÉ SE CREÓ: Para incorporar campos curriculares avanzados (pdaTema, fechas límite).
 * PARÁMETROS: @param {GoogleAppsScript.Spreadsheet.Spreadsheet} [customSpreadsheet] Libro opcional.
 * RESULTADO: {Object} Plan de migración con estado { ok: boolean, status: string, ... }.
 * DEPENDENCIAS: getSpreadsheet_, inspectSheet, applyMissing.
 * HISTORIAL:
 *   - Fecha no registrada: Creación inicial.
 *   - 27/11/2026 (Modificación documental): Documentación obligatoria JSDoc.
 */
function migrateDailyDatabaseSchema_(customSpreadsheet) {
  var ss = customSpreadsheet || (typeof getSpreadsheet_ === 'function' ? getSpreadsheet_() : (typeof obtenerLibro_ === 'function' ? obtenerLibro_() : SpreadsheetApp.getActiveSpreadsheet()));
  if (!ss) throw new Error('No se pudo acceder a la hoja de cálculo de Google Sheets.');

  var targetClases = DB.Clases;
  var targetEvaluaciones = DB.Evaluaciones;

  if (!targetClases || !targetEvaluaciones) {
    throw new Error('DatabaseSchemaError: Definiciones DB.Clases o DB.Evaluaciones no encontradas en el objeto DB.');
  }

  var sheetClases = ss.getSheetByName('Clases');
  var sheetEvaluaciones = ss.getSheetByName('Evaluaciones');

  if (!sheetClases) throw new Error('MigracionError: No se encontró la hoja "Clases" en el libro.');
  if (!sheetEvaluaciones) throw new Error('MigracionError: No se encontró la hoja "Evaluaciones" en el libro.');

  /**
 * TÍTULO: Inspección de encabezados y columnas faltantes en una hoja
 * NOMBRE TÉCNICO: inspectSheet
 * FECHA DE CREACIÓN: Fecha de creación no registrada
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * QUÉ HACE: Compara los encabezados existentes con los esperados y detecta discrepancias.
 * PARA QUÉ SIRVE: Subrutina de análisis para la migración de Registro Diario.
 * POR QUÉ SE CREÓ: Para modularizar la detección de columnas faltantes.
 * PARÁMETROS:
 *   @param {GoogleAppsScript.Spreadsheet.Sheet} sheet Hoja a inspeccionar.
 *   @param {string} sheetName Nombre de la hoja.
 *   @param {Array<string>} targetHeaders Encabezados esperados.
 *   @param {number} minExpectedCols Mínimo de columnas esperado.
 * RESULTADO: {Object} Objeto de diagnóstico.
 * DEPENDENCIAS: Ninguna.
 * HISTORIAL:
 *   - Fecha no registrada: Creación inicial.
 *   - 27/11/2026 (Modificación documental): Documentación obligatoria JSDoc.
 */
function inspectSheet(sheet, sheetName, targetHeaders, minExpectedCols) {
    var lastCol = sheet.getLastColumn();
    var lastRow = sheet.getLastRow();
    if (lastCol === 0 || lastRow === 0) {
      throw new Error('MigracionError: La hoja "' + sheetName + '" está vacía. Debe inicializarse primero con ensureInstalled_.');
    }

    var currentHeaders = sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(function (h) { return String(h || '').trim(); });

    // 1. Verificar duplicados y vacíos
    var seen = {};
    for (var i = 0; i < currentHeaders.length; i++) {
      var h = currentHeaders[i];
      if (!h) {
        throw new Error('MigracionError: La hoja "' + sheetName + '" contiene un encabezado vacío en la columna ' + (i + 1) + '.');
      }
      if (seen[h]) {
        throw new Error('MigracionError: La hoja "' + sheetName + '" contiene la columna duplicada "' + h + '" (columnas ' + seen[h] + ' y ' + (i + 1) + ').');
      }
      seen[h] = i + 1;
    }

    // 2. Verificar longitud mínima (estructura original) y máxima (estructura meta)
    if (currentHeaders.length < minExpectedCols) {
      throw new Error('MigracionError: La hoja "' + sheetName + '" tiene solo ' + currentHeaders.length + ' columnas; se esperaban al menos ' + minExpectedCols + ' columnas de la estructura base.');
    }
    if (currentHeaders.length > targetHeaders.length) {
      throw new Error('MigracionError: La hoja "' + sheetName + '" tiene ' + currentHeaders.length + ' columnas, superando las ' + targetHeaders.length + ' definidas en el esquema.');
    }

    // 3. Verificar orden estricto de columnas existentes
    for (var j = 0; j < currentHeaders.length; j++) {
      if (currentHeaders[j] !== targetHeaders[j]) {
        throw new Error('MigracionError: Discrepancia en la hoja "' + sheetName + '" en la columna ' + (j + 1) + ': se encontró "' + currentHeaders[j] + '", pero se esperaba "' + targetHeaders[j] + '". Orden incompatible.');
      }
    }

    var missingHeaders = targetHeaders.slice(currentHeaders.length);
    return {
      sheet: sheet,
      sheetName: sheetName,
      currentCount: currentHeaders.length,
      targetCount: targetHeaders.length,
      missingHeaders: missingHeaders
    };
  }

  // FASE 1: Inspección y validación pre-vuelo de AMBAS hojas (Sin escrituras)
  var planClases = inspectSheet(sheetClases, 'Clases', targetClases, 14);
  var planEvaluaciones = inspectSheet(sheetEvaluaciones, 'Evaluaciones', targetEvaluaciones, 13);

  // FASE 2: Aplicación atómica de encabezados faltantes al final
  var changesApplied = [];

  /**
 * TÍTULO: Aplicación física de columnas faltantes según plan de migración
 * NOMBRE TÉCNICO: applyMissing
 * FECHA DE CREACIÓN: Fecha de creación no registrada
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * QUÉ HACE: Inserta físicamente los encabezados faltantes al final de la primera fila.
 * PARA QUÉ SIRVE: Ejecuta las adiciones estructurales sin alterar columnas existentes.
 * POR QUÉ SE CREÓ: Para aplicar cambios seguros sin reescritura destructiva de tablas.
 * PARÁMETROS: @param {Object} plan Diagnóstico obtenido por inspectSheet.
 * RESULTADO: {void}
 * DEPENDENCIAS: SpreadsheetApp.
 * HISTORIAL:
 *   - Fecha no registrada: Creación inicial.
 *   - 27/11/2026 (Modificación documental): Documentación obligatoria JSDoc.
 */
function applyMissing(plan) {
    if (plan.missingHeaders.length === 0) return;
    var startCol = plan.currentCount + 1;
    var count = plan.missingHeaders.length;
    var rowValues = [plan.missingHeaders];
    var range = plan.sheet.getRange(1, startCol, 1, count);
    range.setValues(rowValues);
    range.setFontWeight('bold');
    try {
      range.setBackground('#0B315D');
      range.setFontColor('#FFFFFF');
    } catch (e) {}
    changesApplied.push(plan.sheetName + ': se añadieron ' + count + ' columnas al final (' + plan.missingHeaders.join(', ') + ').');
  }

  applyMissing(planClases);
  applyMissing(planEvaluaciones);

  if (changesApplied.length === 0) {
    return {
      ok: true,
      modified: false,
      message: 'Esquema verificado. Ambas hojas ya se encuentran en la estructura correcta; no se requirieron cambios.'
    };
  }

  return {
    ok: true,
    modified: true,
    message: 'Migración completada con éxito:\n- ' + changesApplied.join('\n- ')
  };
}

var AUTHENTIC_MIGRATED_ALUMNOS_COLUMNS = [
  'id', 'nombreCompleto', 'apellidoPaterno', 'apellidoMaterno', 'nombres',
  'fechaNacimiento', 'edad', 'tutorNombre', 'tutorParentesco', 'tutorTelefono',
  'tutorUsuario', 'activo', 'creadoEn', 'actualizadoEn', 'curp', 'sexo',

];

/**
 * Migración puntual, segura e idempotente para el esquema de la hoja Alumnos.
 * 
 * Garantías de integridad y seguridad:
 * 1. Comprueba los encabezados completos y su orden exacto contra el esquema auténtico.
 * 2. Con las 18 columnas originales exactas: permite añadir 'numeroLista' ÚNICAMENTE como columna 19.
 * 3. Con las 19 columnas correctas: devuelve que ya está actualizado (migrated: false), sin escribir.
 * 4. Con columnas reordenadas, duplicadas, vacías, adicionales o numeroLista en una posición
 *    incorrecta (ej. columna 3): RECHAZA la migración con diagnóstico claro y CERO escrituras.
 * 5. No inicializa ni reconstruye una hoja vacía o incompatible.
 * 6. No sobrescribe datos ni elimina columnas para hacer coincidir el esquema.
 * 
 * @param {GoogleAppsScript.Spreadsheet.Spreadsheet} [ss] Hoja de cálculo opcional para pruebas/inyección.
 * @return {Object} Resultado con { ok: boolean, migrated?: boolean, message?: string, error?: string }
 */
/**
 * TÍTULO: Migración e incorporación de la columna número oficial de lista en Alumnos
 * NOMBRE TÉCNICO: migrateAlumnosRosterSchema_
 * FECHA DE CREACIÓN: Fecha de creación no registrada
 * FECHA DE ÚLTIMA MODIFICACIÓN: 27/11/2026
 * QUÉ HACE: Añade la columna 'numeroLista' en la posición 19 de la hoja Alumnos si falta.
 * PARA QUÉ SIRVE: Soporta el número de lista oficial e independiente del orden alfabético.
 * POR QUÉ SE CREÓ: Para cumplir con los lineamientos de acreditación escolar.
 * PARÁMETROS: @param {GoogleAppsScript.Spreadsheet.Spreadsheet} [ss] Libro opcional.
 * RESULTADO: {Object} Resultado con { ok: boolean, yaMigrado: boolean, columna: number }.
 * DEPENDENCIAS: getSpreadsheet_, AUTHENTIC_MIGRATED_ALUMNOS_COLUMNS.
 * HISTORIAL:
 *   - Fecha no registrada: Creación inicial.
 *   - 27/11/2026 (Modificación documental): Documentación obligatoria JSDoc.
 */
function migrateAlumnosRosterSchema_(ss) {
  var targetSs = ss || getSpreadsheet_();
  var sheet = targetSs.getSheetByName('Alumnos');
  if (!sheet) {
    return { ok: false, error: 'Hoja "Alumnos" no encontrada en la hoja de cálculo.' };
  }

  var lastCol = sheet.getLastColumn();
  if (lastCol === 0) {
    return {
      ok: false,
      error: 'La hoja "Alumnos" está vacía o carece de columnas. Esta migración no inicializa ni reconstruye estructuras vacías; requiere la tabla existente.'
    };
  }

  var rawHeaders = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
  var headers = rawHeaders.map(function (h) {
    return String(h || '').trim();
  });

  // 1. Validar encabezados vacíos
  for (var i = 0; i < headers.length; i++) {
    if (!headers[i]) {
      return {
        ok: false,
        error: 'Esquema incompatible: se detectó un encabezado vacío en la columna ' + (i + 1) + '. Operación cancelada sin escrituras.'
      };
    }
  }

  // 2. Validar encabezados duplicados
  var seenHeaders = {};
  for (var d = 0; d < headers.length; d++) {
    var hName = headers[d];
    if (seenHeaders[hName]) {
      return {
        ok: false,
        error: 'Esquema incompatible: encabezado duplicado "' + hName + '" en la columna ' + (d + 1) + '. Operación cancelada sin escrituras.'
      };
    }
    seenHeaders[hName] = true;
  }

  // 3. Evaluar presencia de 'numeroLista' en posición indebida
  var idxNumeroLista = headers.indexOf('numeroLista');
  if (idxNumeroLista !== -1 && idxNumeroLista !== 18) {
    return {
      ok: false,
      error: 'Esquema incompatible: la columna "numeroLista" se encuentra en la posición ' + (idxNumeroLista + 1) + ', pero el estándar oficial requiere que sea estrictamente la columna 19. Operación cancelada sin escrituras.'
    };
  }

  // Caso A: Ya cuenta con exactamente 19 columnas
  if (headers.length === 19) {
    var mismatch19 = [];
    for (var k = 0; k < 19; k++) {
      if (headers[k] !== AUTHENTIC_MIGRATED_ALUMNOS_COLUMNS[k]) {
        mismatch19.push('columna ' + (k + 1) + ': esperada "' + AUTHENTIC_MIGRATED_ALUMNOS_COLUMNS[k] + '", encontrada "' + headers[k] + '"');
      }
    }
    if (mismatch19.length > 0) {
      return {
        ok: false,
        error: 'Esquema incompatible (19 columnas): discrepancias detectadas -> ' + mismatch19.join('; ') + '. Operación cancelada sin escrituras.'
      };
    }
    return {
      ok: true,
      migrated: false,
      message: 'La hoja "Alumnos" ya cuenta con el esquema actualizado de 19 columnas oficiales en el orden exacto. No se realizaron escrituras.'
    };
  }

  // Caso B: Cuenta con exactamente 18 columnas originales
  if (headers.length === 18) {
    var mismatch18 = [];
    for (var j = 0; j < 18; j++) {
      if (headers[j] !== AUTHENTIC_ORIGINAL_ALUMNOS_COLUMNS[j]) {
        mismatch18.push('columna ' + (j + 1) + ': esperada "' + AUTHENTIC_ORIGINAL_ALUMNOS_COLUMNS[j] + '", encontrada "' + headers[j] + '"');
      }
    }
    if (mismatch18.length > 0) {
      return {
        ok: false,
        error: 'Esquema incompatible (18 columnas): discrepancias detectadas -> ' + mismatch18.join('; ') + '. Operación cancelada sin escrituras.'
      };
    }

    // Único caso autorizado para escribir: agregar 'numeroLista' en celda (1, 19)
    sheet.getRange(1, 19).setValue('numeroLista');
    return {
      ok: true,
      migrated: true,
      message: 'Columna "numeroLista" agregada exitosamente como columna 19 al final de la hoja "Alumnos".'
    };
  }

  // Caso C: Cualquier otra cantidad de columnas (menos de 18 o más de 19)
  return {
    ok: false,
    error: 'Esquema incompatible: la hoja tiene ' + headers.length + ' columnas. Se requieren exactamente 18 columnas originales para migrar a 19, o 19 para considerarse actualizada. Operación cancelada sin escrituras.'
  };
}
